package com.dap.dapurhomies.command;

import com.dap.dapurhomies.DapUrHomiesMod;
import com.dap.dapurhomies.Lang;
import com.dap.dapurhomies.ModLifecycle;
import com.dap.dapurhomies.config.ModConfig;
import com.dap.dapurhomies.permission.PlayerPermissionManager;
import com.mojang.brigadier.CommandDispatcher;
import net.minecraft.command.argument.EntityArgumentType;
import net.minecraft.server.command.CommandManager;
import net.minecraft.server.command.ServerCommandSource;
import net.minecraft.server.network.ServerPlayerEntity;
import net.minecraft.text.Text;

public class DapCommands {
    public static void register(CommandDispatcher<ServerCommandSource> dispatcher) {
        var dapRoot = CommandManager.literal("dap");

        // /dap on [player]
        dapRoot.then(CommandManager.literal("on")
            .executes(c -> {
                ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                if (PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) {
                    if (DapUrHomiesMod.isModEnabled()) Lang.alreadyEnabled(p);
                    else { DapUrHomiesMod.setModEnabled(true); Lang.modEnabled(p); }
                } else {
                    if (com.dap.dapurhomies.util.AdminManager.isForceDisabled(p.getUuid())) {
                        Lang.interactionBlocked(p, "An admin disabled this for you!", "管理员已关闭你的交互功能！");
                        return 0;
                    }
                    PlayerPermissionManager.set(p, "enabled", true);
                    Lang.playerEnabled(p);
                }
                return 1;
            })
            .then(CommandManager.argument("player", EntityArgumentType.player())
                .executes(c -> {
                    ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                    if (!PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) {
                        Lang.noPermission(p); return 0;
                    }
                    ServerPlayerEntity t = EntityArgumentType.getPlayer(c, "player");
                    com.dap.dapurhomies.util.AdminManager.setForceDisabled(t.getUuid(), false);
                    PlayerPermissionManager.set(t, "enabled", true);
                    Lang.playerEnabledByAdmin(p, t);
                    return 1;
                })));

        // /dap off [player]
        dapRoot.then(CommandManager.literal("off")
            .executes(c -> {
                ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                if (PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) {
                    if (!DapUrHomiesMod.isModEnabled()) Lang.alreadyDisabled(p);
                    else ModLifecycle.disableModAndCleanup(p);
                } else {
                    PlayerPermissionManager.set(p, "enabled", false);
                    Lang.playerDisabled(p);
                }
                return 1;
            })
            .then(CommandManager.argument("player", EntityArgumentType.player())
                .executes(c -> {
                    ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                    if (!PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) {
                        Lang.noPermission(p); return 0;
                    }
                    ServerPlayerEntity t = EntityArgumentType.getPlayer(c, "player");
                    com.dap.dapurhomies.util.AdminManager.setForceDisabled(t.getUuid(), true);
                    PlayerPermissionManager.set(t, "enabled", false);
                    Lang.playerDisabledByAdmin(p, t);
                    return 1;
                })));

        // /dap status [player]
        dapRoot.then(CommandManager.literal("status")
            .executes(c -> {
                ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                Lang.modStatus(p, DapUrHomiesMod.isModEnabled());
                Lang.playerStatus(p, PlayerPermissionManager.get(p));
                return 1;
            })
            .then(CommandManager.argument("player", EntityArgumentType.player())
                .executes(c -> {
                    ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                    if (!PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) {
                        Lang.noPermission(p); return 0;
                    }
                    ServerPlayerEntity t = EntityArgumentType.getPlayer(c, "player");
                    Lang.modStatus(p, DapUrHomiesMod.isModEnabled());
                    Lang.playerStatus(p, t, PlayerPermissionManager.get(t));
                    return 1;
                })));

        // /dap reload
        dapRoot.then(CommandManager.literal("reload")
            .executes(c -> {
                ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                if (!PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) {
                    Lang.noPermission(p); return 0;
                }
                try {
                    ModConfig.load();
                    Lang.configReloaded(p);
                } catch (Exception e) {
                    Lang.configReloadFailed(p);
                }
                return 1;
            }));

        // /dap admincooldown on|off
        dapRoot.then(CommandManager.literal("admincooldown")
            .then(CommandManager.literal("on")
                .executes(c -> {
                    ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                    if (!PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) {
                        Lang.noPermission(p); return 0;
                    }
                    ModConfig.get().adminBypassCooldown = true;
                    ModConfig.save();
                    Lang.adminCooldownToggled(p, true);
                    return 1;
                }))
            .then(CommandManager.literal("off")
                .executes(c -> {
                    ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                    if (!PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) {
                        Lang.noPermission(p); return 0;
                    }
                    ModConfig.get().adminBypassCooldown = false;
                    ModConfig.save();
                    Lang.adminCooldownToggled(p, false);
                    return 1;
                })));

        String[] types = {"dap", "grab", "push", "kick", "hug", "slap", "clap", "danse"};
        for (String type : types) {
            var typeLit = CommandManager.literal(type);
            typeLit.then(CommandManager.literal("on")
                .executes(c -> toggleType(c.getSource().getPlayerOrThrow(), type, true, null))
                .then(CommandManager.argument("player", EntityArgumentType.player())
                    .executes(c -> {
                        ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                        if (!PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) {
                            Lang.noPermission(p); return 0;
                        }
                        return toggleType(p, type, true, EntityArgumentType.getPlayer(c, "player"));
                    })));
            typeLit.then(CommandManager.literal("off")
                .executes(c -> toggleType(c.getSource().getPlayerOrThrow(), type, false, null))
                .then(CommandManager.argument("player", EntityArgumentType.player())
                    .executes(c -> {
                        ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                        if (!PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) {
                            Lang.noPermission(p); return 0;
                        }
                        return toggleType(p, type, false, EntityArgumentType.getPlayer(c, "player"));
                    })));
            dapRoot.then(typeLit);
        }

        // /dap ban <player> <type|all> <minutes>
        dapRoot.then(CommandManager.literal("ban")

            .then(CommandManager.argument("player", EntityArgumentType.player())
            .then(CommandManager.argument("type", com.mojang.brigadier.arguments.StringArgumentType.word())
            .then(CommandManager.argument("minutes", com.mojang.brigadier.arguments.IntegerArgumentType.integer(1))
                .executes(c -> {
                    ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                    if (!PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) {
                        Lang.noPermission(p); return 0;
                    }
                    ServerPlayerEntity t = EntityArgumentType.getPlayer(c, "player");
                    String type = com.mojang.brigadier.arguments.StringArgumentType.getString(c, "type");
                    if (!com.dap.dapurhomies.util.AdminManager.isValidType(type)) {
                        Lang.invalidCooldownType(p); return 0;
                    }
                    int min = com.mojang.brigadier.arguments.IntegerArgumentType.getInteger(c, "minutes");
                    com.dap.dapurhomies.util.AdminManager.ban(t.getUuid(), type, min * 1200L);
                    c.getSource().sendFeedback(() -> Text.literal("Banned " + t.getName().getString() + " from " + type + " for " + min + "min"), false);
                    return 1;
                })))));

        // /dap unban <player> <type|all>
        dapRoot.then(CommandManager.literal("unban")

            .then(CommandManager.argument("player", EntityArgumentType.player())
            .then(CommandManager.argument("type", com.mojang.brigadier.arguments.StringArgumentType.word())
                .executes(c -> {
                    ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                    if (!PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) {
                        Lang.noPermission(p); return 0;
                    }
                    ServerPlayerEntity t = EntityArgumentType.getPlayer(c, "player");
                    String type = com.mojang.brigadier.arguments.StringArgumentType.getString(c, "type");
                    if (!com.dap.dapurhomies.util.AdminManager.isValidType(type)) {
                        Lang.invalidCooldownType(p); return 0;
                    }
                    com.dap.dapurhomies.util.AdminManager.unban(t.getUuid(), type);
                    c.getSource().sendFeedback(() -> Text.literal("Unbanned " + t.getName().getString()), false);
                    return 1;
                }))));

        // /dap cooldown ms|set ...
        dapRoot.then(CommandManager.literal("cooldown")
            .then(CommandManager.literal("ms")
            .then(CommandManager.argument("type", com.mojang.brigadier.arguments.StringArgumentType.string())
                .suggests((ctx, builder) -> {
                    String[] suggest = {"dap", "highFive", "grab", "push", "kick", "hug", "slap"};
                    for (String t : suggest) builder.suggest(t);
                    return builder.buildFuture();
                })
                .then(CommandManager.argument("ms", com.mojang.brigadier.arguments.IntegerArgumentType.integer(0, 300000))
                    .executes(c -> {
                        ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                        if (!PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) {
                            Lang.noPermission(p); return 0;
                        }
                        String type = com.mojang.brigadier.arguments.StringArgumentType.getString(c, "type");
                        int ms = com.mojang.brigadier.arguments.IntegerArgumentType.getInteger(c, "ms");
                        switch (type) {
                            case "dap" -> ModConfig.get().dapCooldownMs = ms;
                            case "highFive" -> ModConfig.get().highFiveCooldownMs = ms;
                            case "grab" -> ModConfig.get().grabCooldownMs = ms;
                            case "push" -> ModConfig.get().pushCooldownMs = ms;
                            case "kick" -> ModConfig.get().kickCooldownMs = ms;
                            case "hug" -> ModConfig.get().hugCooldownMs = ms;
                            case "slap" -> ModConfig.get().slapCooldownMs = ms;
                            default -> { Lang.invalidCooldownType(p); return 0; }
                        }
                        ModConfig.save();
                        Lang.cooldownSet(p, type, ms);
                        return 1;
                    }))))
            .then(CommandManager.literal("set")
            .then(CommandManager.argument("type", com.mojang.brigadier.arguments.StringArgumentType.word())
            .then(CommandManager.argument("seconds", com.mojang.brigadier.arguments.IntegerArgumentType.integer(1))
                .executes(c -> {
                    ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                    if (!PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) {
                        Lang.noPermission(p); return 0;
                    }
                    String type = com.mojang.brigadier.arguments.StringArgumentType.getString(c, "type");
                    if (!com.dap.dapurhomies.util.AdminManager.isValidType(type) || "all".equals(type)) {
                        Lang.invalidCooldownType(p); return 0;
                    }
                    long ticks = com.mojang.brigadier.arguments.IntegerArgumentType.getInteger(c, "seconds") * 20L;
                    com.dap.dapurhomies.util.AdminManager.setGlobalCooldown(type, ticks);
                    c.getSource().sendFeedback(() -> Text.literal("Global " + type + " cooldown = " + (ticks/20) + "s"), false);
                    return 1;
                })
                .then(CommandManager.argument("player", EntityArgumentType.player())
                    .executes(c -> {
                        ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                        if (!PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) { Lang.noPermission(p); return 0; }
                        String type = com.mojang.brigadier.arguments.StringArgumentType.getString(c, "type");
                        if (!com.dap.dapurhomies.util.AdminManager.isValidType(type) || "all".equals(type)) {
                            Lang.invalidCooldownType(p); return 0;
                        }
                        long ticks = com.mojang.brigadier.arguments.IntegerArgumentType.getInteger(c, "seconds") * 20L;
                        ServerPlayerEntity t = EntityArgumentType.getPlayer(c, "player");
                        com.dap.dapurhomies.util.AdminManager.setPlayerCooldown(t.getUuid(), type, ticks);
                        c.getSource().sendFeedback(() -> Text.literal(t.getName().getString() + " " + type + " cooldown = " + (ticks/20) + "s"), false);
                        return 1;
                    }))))));

        // /dap sound <ticks>
        dapRoot.then(CommandManager.literal("sound")

            .then(CommandManager.argument("ticks", com.mojang.brigadier.arguments.IntegerArgumentType.integer(1))
                .executes(c -> {
                    ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                    if (!PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) { Lang.noPermission(p); return 0; }
                    ModConfig.get().highFiveSoundTicks = com.mojang.brigadier.arguments.IntegerArgumentType.getInteger(c, "ticks");
                    ModConfig.save();
                    c.getSource().sendFeedback(() -> Text.literal("HighFive sound = " + ModConfig.get().highFiveSoundTicks + " ticks"), false);
                    return 1;
                })));

        // /dap chargedmode <0|1>
        dapRoot.then(CommandManager.literal("chargedmode")
            .then(CommandManager.argument("mode", com.mojang.brigadier.arguments.IntegerArgumentType.integer(0, 1))
                .executes(c -> {
                    ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                    if (!PlayerPermissionManager.hasPermission(p, "dap-ur-homies.admin")) {
                        Lang.noPermission(p); return 0;
                    }
                    int mode = com.mojang.brigadier.arguments.IntegerArgumentType.getInteger(c, "mode");
                    ModConfig.get().chargedDapMode = mode;
                    ModConfig.save();
                    Lang.chargedModeSet(p, mode);
                    return 1;
                })));

        // Default /dap
        dapRoot.executes(c -> {
            c.getSource().sendFeedback(() -> Lang.t(
                "\u00a7e[Dap] \u00a7f/dap on|off [player] | /dap <type> on|off [player] | /dap status [player]",
                "\u00a7e[\u78b0\u62f3] \u00a7f/dap on|off [\u73a9\u5bb6] | /dap <\u7c7b\u578b> on|off [\u73a9\u5bb6] | /dap status [\u73a9\u5bb6]"
            ), false);
            return 1;
        });

        dispatcher.register(dapRoot);
    }

    private static int toggleType(ServerPlayerEntity sender, String type, boolean value, ServerPlayerEntity target) {
        ServerPlayerEntity actualTarget = (target != null) ? target : sender;
        PlayerPermissionManager.set(actualTarget, type, value);
        if (target != null && !sender.equals(target)) {
            Lang.typeToggledByAdmin(sender, target, type, value);
        } else {
            Lang.typeToggled(sender, type, value);
        }
        return 1;
    }
}

public class InteractionCommands {

        }));

        // /push
        dispatcher.register(CommandManager.literal("push").executes(c -> {
            ServerPlayerEntity p = c.getSource().getPlayerOrThrow();

            if (!PlayerPermissionManager.canInitiate(p, "clap")) { Lang.playerTypeDisabled(p, "clap"); return 0; }
            SimpleInteractions.onClap(p);
            return 1;
        }));

        // /mariojump
        dispatcher.register(CommandManager.literal("mariojump").executes(c -> {
            ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
            if (!DapUrHomiesMod.isModEnabled()) { Lang.modStatus(p, false); return 0; }
            SimpleInteractions.onMarioJump(p);
            return 1;
        }));

        // /daplang self <en|zh|global>
        var daplangRoot = CommandManager.literal("daplang");
        daplangRoot.then(CommandManager.literal("self")
            .then(CommandManager.argument("lang", com.mojang.brigadier.arguments.StringArgumentType.string())
                .suggests((ctx, builder) -> {
                    builder.suggest("en"); builder.suggest("zh"); builder.suggest("global");
                    return builder.buildFuture();
                })
                .executes(c -> {
                    ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
                    String lang = com.mojang.brigadier.arguments.StringArgumentType.getString(c, "lang").toLowerCase();
                    if (!lang.equals("en") && !lang.equals("zh") && !lang.equals("global")) {
                        p.sendMessage(Text.literal("\u00a7c[Dap] Invalid language! Use: en, zh, global"), false);
                        return 0;
                    }
                    PlayerPermissionManager.setLang(p.getUuid(), lang);
                    Lang.langSetSelf(p, lang);
                    return 1;
                })));
        // /daplang global <en|zh> (admin only)
        daplangRoot.then(CommandManager.literal("global")
            .then(CommandManager.argument("lang", com.mojang.brigadier.arguments.StringArgumentType.string())
                .suggests((ctx, builder) -> {
                    builder.suggest("en"); builder.suggest("zh");
                    return builder.buildFuture();

                        return 0;
                    }
                    Lang.setGlobalLanguage(lang.equals("zh") ? Lang.Language.ZH : Lang.Language.EN);
                    Lang.langSetGlobal(p, lang);
                    return 1;

        daplangRoot.executes(c -> {
            ServerPlayerEntity p = c.getSource().getPlayerOrThrow();
            Lang.langCurrent(p);
            return 1;
        });
        dispatcher.register(daplangRoot);

package com.dap.dapurhomies.compat;

import com.dap.dapurhomies.DapUrHomiesMod;
import com.dap.dapurhomies.permission.PlayerPermissionManager;
import net.fabricmc.loader.api.FabricLoader;
import net.minecraft.server.network.ServerPlayerEntity;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.lang.reflect.Method;
import java.util.HashSet;
import java.util.Set;

public class DanseCompat {
    private static final Logger LOGGER = LoggerFactory.getLogger("dap-ur-homies");
    private static final boolean DANSE_LOADED = FabricLoader.getInstance().isModLoaded("danse");
    private static Method onStartMethod;
    private static Method onStopMethod;
    private static Method getModelMethod;

    private static final Set<String> FAILED_ANIMATIONS = new HashSet<>();

    static {
        if (DANSE_LOADED) {
            try {
                Class<?> gestureController = Class.forName("de.tomalbrc.danse.GestureController");
                onStartMethod = gestureController.getMethod("onStart", ServerPlayerEntity.class, String.class);
                onStopMethod = gestureController.getMethod("onStop", ServerPlayerEntity.class);
                try {
                    Class<?> registry = Class.forName("de.tomalbrc.danse.registry.PlayerModelRegistry");
                    getModelMethod = registry.getMethod("getModel", String.class);
                } catch (Exception e) {
                    LOGGER.warn("[DapUrHomies] Danse animation registry is unavailable; custom animation names will be used unchanged.");
                }
                LOGGER.info("[DapUrHomies] Danse integration enabled.");
            } catch (Exception e) {
                LOGGER.warn("[DapUrHomies] Danse detected but GestureController API is incompatible: " + e.getMessage());
            }
        }
    }

    public static boolean isDanseLoaded() {
        return DANSE_LOADED && onStartMethod != null;
    }

    private static boolean canPlayPair(ServerPlayerEntity initiator, ServerPlayerEntity target) {
        return isDanseLoaded()
                && DapUrHomiesMod.isModEnabled()
                && PlayerPermissionManager.canInitiate(initiator, "danse")
                && PlayerPermissionManager.canReceive(target, "danse");
    }

    private static String resolveAnimation(String... candidates) {
        if (getModelMethod == null) return candidates[0];
        for (String candidate : candidates) {
            try {
                // 优先使用服务器安装的同名自定义动画，缺失时使用 Danse 内置动画。
                if (getModelMethod.invoke(null, candidate) != null) return candidate;
            } catch (Exception e) {
                if (FAILED_ANIMATIONS.add(candidate)) {
                    LOGGER.warn("[DapUrHomies] Failed to query Danse animation '{}': {}", candidate, e.getMessage());
                }
                return null;
            }
        }
        if (FAILED_ANIMATIONS.add(candidates[0])) {
            LOGGER.warn("[DapUrHomies] No registered Danse animation found for '{}'. Tried: {}",
                    candidates[0], String.join(", ", candidates));
        }
        return null;
    }

    public static void playAnimation(ServerPlayerEntity player, String... candidates) {
        if (!isDanseLoaded() || player == null || candidates.length == 0) return;
        String animationName = resolveAnimation(candidates);
        if (animationName == null) return;
        try {
            onStartMethod.invoke(null, player, animationName);
        } catch (Exception e) {
            if (FAILED_ANIMATIONS.add(animationName)) {
                LOGGER.warn("[DapUrHomies] Failed to play Danse animation '{}'.", animationName, e);
            }
        }
    }

    public static void stopAnimation(ServerPlayerEntity player) {
        if (!isDanseLoaded() || onStopMethod == null || player == null) return;
        try {
            onStopMethod.invoke(null, player);
        } catch (Exception e) {
            LOGGER.warn("[DapUrHomies] Failed to stop Danse animation.", e);
        }
    }

    public static void playDap(ServerPlayerEntity p1, ServerPlayerEntity p2) {
        if (!canPlayPair(p1, p2)) return;
        playAnimation(p1, "dap", "dab");
        playAnimation(p2, "dap", "dab");
    }

    public static void playHighFive(ServerPlayerEntity p1, ServerPlayerEntity p2) {
        if (!canPlayPair(p1, p2)) return;
        playAnimation(p1, "highfive", "high_five", "wave");
        playAnimation(p2, "highfive", "high_five", "wave");
    }

    public static void playHug(ServerPlayerEntity p1, ServerPlayerEntity p2) {
        if (!canPlayPair(p1, p2)) return;
        playAnimation(p1, "hug", "fold");
        playAnimation(p2, "hug", "fold");
    }

    public static void playGrab(ServerPlayerEntity holder, ServerPlayerEntity victim) {
        if (!canPlayPair(holder, victim)) return;
        playAnimation(victim, "grabbed", "fall");
    }

    public static void playYeet(ServerPlayerEntity victim) {
        if (!isDanseLoaded() || !PlayerPermissionManager.canInitiate(victim, "danse")) return;
        playAnimation(victim, "yeet", "ascend");
    }

    public static void playKick(ServerPlayerEntity kicker, ServerPlayerEntity victim) {
        if (!canPlayPair(kicker, victim)) return;
        playAnimation(kicker, "kick", "bits");
        playAnimation(victim, "kicked", "fall");
    }

    public static void playPush(ServerPlayerEntity pusher, ServerPlayerEntity victim) {
        if (!canPlayPair(pusher, victim)) return;
        playAnimation(pusher, "push", "bits");
        playAnimation(victim, "pushed", "fall");
    }

    public static void playSlap(ServerPlayerEntity slapper, ServerPlayerEntity victim) {
        if (!canPlayPair(slapper, victim)) return;
        playAnimation(slapper, "slap", "facepalm");
        playAnimation(victim, "slapped", "facepalm");
    }

    public static void playClap(ServerPlayerEntity player) {
        if (!isDanseLoaded() || !PlayerPermissionManager.canInitiate(player, "danse")) return;
        playAnimation(player, "clap", "wave");
    }

    public static void playMarioJump(ServerPlayerEntity player) {
        if (!isDanseLoaded() || !PlayerPermissionManager.canInitiate(player, "danse")) return;
        playAnimation(player, "mario_jump", "ascend");
    }

            try {
                Object dataStore = GP_GET_DATA_STORE.invoke(gpInstance);
                Object claim = GP_GET_CLAIM_AT.invoke(dataStore, pos, false, null);
                if (claim != null) {
                    Object result = GP_CLAIM_ALLOW_BUILD.invoke(claim, initiator);
                    if (result != null) return false;
                }
            } catch (Throwable ignored) {}
        }

        if (landsApi != null) {

                if (LANDS_GET_AREA_WBP != null) {
                    area = LANDS_GET_AREA_WBP.invoke(landsApi, initiator.getEntityWorld(), pos);
                } else if (LANDS_GET_AREA_BP != null) {

            try {
                Object result = FLAN_CAN_INTERACT.invoke(null, initiator, pos, flanPermission);
                if (result instanceof Boolean && !(Boolean) result) return false;
            } catch (Throwable ignored) {}
        }

        return true;
    }

import net.minecraft.entity.attribute.EntityAttributeInstance;

import net.minecraft.entity.attribute.EntityAttributes;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.network.ServerPlayerEntity;
import net.minecraft.util.Identifier;

/**
 * Applies speed slowdown to players who are holding/grabbing other players.
 * Ported from original Dap-ur-homie project.
 */
public class CarryingSlowdown {

    public static final float CARRY_SPEED_MULTIPLIER = 0.6f;
    private static final double SLOWDOWN_AMOUNT = -(1.0 - CARRY_SPEED_MULTIPLIER) * 0.1;
    private static final Identifier MODIFIER_ID = Identifier.of("dap-ur-homies", "carrying_slowdown");

    public static void tick(MinecraftServer server) {
        for (ServerPlayerEntity player : server.getPlayerManager().getPlayerList()) {
            updateSlowdown(player);
        }
    }

    private static void updateSlowdown(ServerPlayerEntity player) {
        boolean isCarrying = GrabYeetHandler.isHolding(player);

        EntityAttributeInstance speedAttr = player.getAttributeInstance(EntityAttributes.MOVEMENT_SPEED);
        if (speedAttr == null) return;

        var existingModifier = speedAttr.getModifier(MODIFIER_ID);

        if (isCarrying) {
            if (existingModifier == null) {
                speedAttr.addTemporaryModifier(new EntityAttributeModifier(
                        MODIFIER_ID,
                        SLOWDOWN_AMOUNT,
                        EntityAttributeModifier.Operation.ADD_VALUE
                ));
            }
        } else {
            if (existingModifier != null) {
                speedAttr.removeModifier(MODIFIER_ID);
            }
        }
    }
}

package com.dap.dapurhomies.interaction;

public class DapHandler {
    private static final Map<UUID, Long> pendingHighFive = new ConcurrentHashMap<>();
    private static final Map<UUID, Long> chargeStart = new ConcurrentHashMap<>();
    private static final Map<UUID, Integer> chargeLevels = new ConcurrentHashMap<>();
    private static final Map<UUID, UUID> chargeTarget = new ConcurrentHashMap<>();
    private static final long HIGH_FIVE_WINDOW_TICKS = TickClock.msToTicks(800);
    private static final long[] CHARGE_THRESHOLDS = {0, 10, 20, 30, 40, 50};
    private static final long MAX_CHARGE_TICKS = TickClock.msToTicks(3000);
    private static final long CHARGE_TIMEOUT_TICKS = 200;
    private static final long CYCLE_CHARGE_TICKS = TickClock.msToTicks(8000); // 8s for cycle mode
    private static final long CYCLE_CHARGE_TIMEOUT_TICKS = 400; // 20s for cycle mode

    // Perfect Dap: a max-level release stays "answerable" for this long (1.5s).
    // Replaces the old 2-tick charge-START sync requirement which was humanly impossible.
    private static final long PERFECT_WINDOW_TICKS = TickClock.msToTicks(1500);
    private static final Map<UUID, PendingPerfect> pendingPerfect = new ConcurrentHashMap<>();

    private record PendingPerfect(UUID targetId, int level, long releaseTick) {}

    private static final int SPEED_HISTORY_TICKS = 30;
    private static final double SPEED_TIER_1 = 5.0, SPEED_TIER_2 = 7.5, SPEED_TIER_3 = 12.0;
    private static final Map<UUID, LinkedList<Double>> speedHistory = new ConcurrentHashMap<>();

    private static final Map<UUID, PendingHighFive> pendingEffects = new ConcurrentHashMap<>();
    private static final long HIT_EFFECT_DELAY_TICKS = TickClock.msToTicks(100);

    private static class PendingHighFive {
        final ServerPlayerEntity p1, p2;
        final Vec3d pos;
        final int tier;
        final long effectTime;
        PendingHighFive(ServerPlayerEntity p1, ServerPlayerEntity p2, Vec3d pos, int tier, long effectTime) {
            this.p1 = p1; this.p2 = p2; this.pos = pos; this.tier = tier; this.effectTime = effectTime;
        }
    }

    public static void tick(MinecraftServer server) {
        long now = TickClock.now();
        for (ServerPlayerEntity player : server.getPlayerManager().getPlayerList()) {
            tickSpeedHistory(player);
            tickCharge(player);
        }
        Iterator<Map.Entry<UUID, PendingHighFive>> it = pendingEffects.entrySet().iterator();
        while (it.hasNext()) {
            Map.Entry<UUID, PendingHighFive> e = it.next();
            if (now >= e.getValue().effectTime) {
                PendingHighFive fx = e.getValue();
                boolean online = server.getPlayerManager().getPlayer(fx.p1.getUuid()) != null
                        && server.getPlayerManager().getPlayer(fx.p2.getUuid()) != null;
                if (online) {
                    DapVisualEffects.executeHighFiveEffects(fx.p1, fx.p2, fx.pos, fx.tier);
                }
                it.remove();
            }
        }
    }

    private static void tickSpeedHistory(ServerPlayerEntity player) {
        UUID id = player.getUuid();
        double speed = player.getVelocity().length() * 20.0;
        LinkedList<Double> history = speedHistory.computeIfAbsent(id, k -> new LinkedList<>());
        history.addLast(speed);
        while (history.size() > SPEED_HISTORY_TICKS) history.removeFirst();
    }

    private static double getMaxRecentSpeed(UUID playerId) {
        LinkedList<Double> history = speedHistory.get(playerId);
        if (history == null || history.isEmpty()) return 0.0;
        double max = 0.0;
        for (Double s : history) if (s > max) max = s;
        return max;
    }

    public static ActionResult startCharge(ServerPlayerEntity sp, ServerPlayerEntity target) {
        if (sp == null || target == null) return ActionResult.PASS;
        if (!ModConfig.get().enableDap) return ActionResult.PASS;
        if (sp.getEntityWorld() != target.getEntityWorld()) return ActionResult.PASS;
        if (sp.distanceTo(target) > ModConfig.get().dapRange) return ActionResult.PASS;
        if (!PlayerPermissionManager.canInitiate(sp, "dap")) {
            Lang.interactionBlocked(sp, "Dap is disabled for you!", "你的碰拳功能已关闭！");
            return ActionResult.FAIL;
        }
        if (InteractionCooldowns.rejectIfBlocked(sp, "dap")) return ActionResult.FAIL;
        if (!PlayerPermissionManager.hasPermission(sp, "dap-ur-homies.admin")
                && !PlayerPermissionManager.canReceive(target, "dap")) {
            Lang.targetDisabledType(sp, target, "dap");
            return ActionResult.FAIL;
        }
        if (DapSessionManager.isInSession(sp.getUuid()) || DapSessionManager.isInSession(target.getUuid()))
            return ActionResult.PASS;

        // Mode 1 (cycle-charge): if already charging, release instead
        // In cycle mode, a second click releases the current charge; this must precede
        // the general charging check below, which would otherwise return PASS first.
        if (ModConfig.get().chargedDapMode == 1 && isCharging(sp.getUuid())) {
            releaseCharge(sp);
            return ActionResult.SUCCESS;
        }

        if (isCharging(sp.getUuid())) return ActionResult.PASS;
        // Mutual charging is allowed; reject only when the target is charging someone else.
        // Perfect Dap requires both players to be able to charge each other.
        UUID targetChargeTarget = chargeTarget.get(target.getUuid());
        if (targetChargeTarget != null && !targetChargeTarget.equals(sp.getUuid()))
            return ActionResult.PASS;

        chargeStart.put(sp.getUuid(), TickClock.now());
        chargeLevels.put(sp.getUuid(), 0);
        chargeTarget.put(sp.getUuid(), target.getUuid());
        Lang.chargeStart(sp);
        return ActionResult.SUCCESS;
    }

    public static ActionResult handleNormalDap(ServerPlayerEntity sp, ServerPlayerEntity target) {
        if (sp == null || target == null) return ActionResult.PASS;
        if (!ModConfig.get().enableDap) return ActionResult.PASS;
        if (sp.getEntityWorld() != target.getEntityWorld()) return ActionResult.PASS;
        if (sp.distanceTo(target) > ModConfig.get().dapRange) return ActionResult.PASS;
        if (!PlayerPermissionManager.canInitiate(sp, "dap")) {
            Lang.interactionBlocked(sp, "Dap is disabled for you!", "你的碰拳功能已关闭！");
            return ActionResult.FAIL;
        }
        long now = TickClock.now();
        if (InteractionCooldowns.rejectIfBlocked(sp, "dap")) return ActionResult.FAIL;
        if (!PlayerPermissionManager.hasPermission(sp, "dap-ur-homies.admin")
                && !PlayerPermissionManager.canReceive(target, "dap")) {
            Lang.targetDisabledType(sp, target, "dap");
            return ActionResult.FAIL;
        }
        if (DapSessionManager.isInSession(sp.getUuid()) || DapSessionManager.isInSession(target.getUuid()))
            return ActionResult.PASS;
        if (isCharging(sp.getUuid()) || isCharging(target.getUuid()))
            return ActionResult.PASS;
        UUID sid = sp.getUuid(), tid = target.getUuid();
        Long pending = pendingHighFive.get(tid);
        boolean highFiveAllowed = ModConfig.get().enableHighFive
                && sp.distanceTo(target) <= ModConfig.get().highFiveRange;
        if (highFiveAllowed && pending != null && now - pending < HIGH_FIVE_WINDOW_TICKS) {
            if (!PlayerPermissionManager.canInitiate(sp, "highFive")) {
                Lang.interactionBlocked(sp, "High-five is disabled for you!", "你的击掌功能已关闭！");
                return ActionResult.FAIL;
            }
            if (!PlayerPermissionManager.hasPermission(sp, "dap-ur-homies.admin")
                    && !PlayerPermissionManager.canReceive(target, "highFive")) {
                Lang.targetDisabledType(sp, target, "highFive");
                return ActionResult.FAIL;
            }
            if (InteractionCooldowns.rejectIfBlocked(sp, "highFive")) return ActionResult.FAIL;
            executeHighFive(sp, target);
            InteractionCooldowns.applyShared(sid, tid, "highFive", ModConfig.get().highFiveCooldownMs);
            pendingHighFive.remove(tid); pendingHighFive.remove(sid);
            return ActionResult.SUCCESS;
        }
        pendingHighFive.put(sid, now);
        executeDap(sp, target, 1);
        InteractionCooldowns.apply(sid, "dap", ModConfig.get().dapCooldownMs);
        return ActionResult.SUCCESS;
    }

    public static void tickCharge(ServerPlayerEntity player) {
        if (player == null) return;
        UUID id = player.getUuid();
        Long start = chargeStart.get(id);
        if (start == null) return;
        long elapsed = TickClock.now() - start;

        int mode = ModConfig.get().chargedDapMode;
        if (mode == 0) {
            // Mode 0: hold-charge (original) — release when not sneaking or exceeded max charge time
            if (!player.isSneaking() || elapsed > MAX_CHARGE_TICKS) { releaseCharge(player); return; }
            int newLevel = 0;
            for (int i = CHARGE_THRESHOLDS.length - 1; i >= 0; i--) {
                if (elapsed >= CHARGE_THRESHOLDS[i]) { newLevel = i; break; }
            }
            Integer oldLevel = chargeLevels.get(id);
            if (oldLevel == null || newLevel != oldLevel) {
                chargeLevels.put(id, newLevel);
                Lang.chargeLevel(player, newLevel);
            }
        } else {
            // Mode 1: cycle-charge — 8s session with looping levels 0->5->0 (triangular wave)
            if (elapsed > CYCLE_CHARGE_TICKS) { releaseCharge(player); return; }
            double progress = (double) elapsed / CYCLE_CHARGE_TICKS;
            double tri = progress < 0.5 ? progress * 2.0 : (1.0 - progress) * 2.0;
            int newLevel = (int) Math.round(tri * 5.0);
            newLevel = Math.max(0, Math.min(5, newLevel));
            Integer oldLevel = chargeLevels.get(id);
            if (oldLevel == null || newLevel != oldLevel) {
                chargeLevels.put(id, newLevel);
                Lang.chargeLevel(player, newLevel);
            }
        }

        if (TickClock.now() % 4 == 0) {
            ServerWorld w = (ServerWorld) player.getEntityWorld();
            if (w != null) {
                Vec3d pos = player.getEntityPos().add(0, 1, 0);
                w.spawnParticles(ParticleTypes.CRIT, pos.x, pos.y, pos.z, 2, 0.3, 0.3, 0.3, 0.05);
            }
        }
    }

    public static boolean isCharging(UUID id) { return chargeStart.containsKey(id); }

    /** Abort a charge without firing any effect (used when another interaction wins). */
    public static void cancelCharge(UUID id) {
        if (id == null) return;
        chargeStart.remove(id);
        chargeLevels.remove(id);
        chargeTarget.remove(id);
    }

    public static void releaseCharge(ServerPlayerEntity player) {
        if (player == null) return;
        UUID id = player.getUuid();
        Long start = chargeStart.remove(id);
        Integer level = chargeLevels.remove(id);
        UUID targetId = chargeTarget.remove(id);
        if (start == null) return;
        int powerLevel = (level != null) ? level : 0;
        ServerPlayerEntity target = null;
        if (targetId != null) {
            var server = ((ServerWorld) player.getEntityWorld()).getServer();
            if (server != null) target = server.getPlayerManager().getPlayer(targetId);
        }
        if (target == null || target.getEntityWorld() != player.getEntityWorld()
                || player.distanceTo(target) > ModConfig.get().dapRange) {
            Lang.chargeCancelled(player);
            InteractionCooldowns.apply(id, "dap", ModConfig.get().dapCooldownMs);
            return;
        }

        int maxLevel = CHARGE_THRESHOLDS.length - 1;
        long now = TickClock.now();

        // Perfect Dap (v0.6.0 rework): I must release at max level, and the partner qualifies by EITHER
        //   a) still max-charging AT ME right now, OR
        //   b) having max-released AT ME within PERFECT_WINDOW_TICKS (pending-release window).
        // The old Mode-0 rule compared charge-START times with a 2-tick (0.1s) tolerance,
        // which was humanly impossible to satisfy.
        Long partnerStart = chargeStart.get(target.getUuid());
        Integer partnerLevel = chargeLevels.get(target.getUuid());
        UUID partnerTarget = chargeTarget.get(target.getUuid());
        boolean partnerIsChargingMe = partnerTarget != null && partnerTarget.equals(id);
        boolean partnerCurrentlyMax = partnerIsChargingMe && partnerStart != null
                && partnerLevel != null && partnerLevel >= maxLevel;

        PendingPerfect pp = pendingPerfect.get(target.getUuid());
        boolean partnerRecentlyReleasedMax = pp != null && pp.targetId().equals(id)
                && pp.level() >= maxLevel && now - pp.releaseTick() <= PERFECT_WINDOW_TICKS;

        boolean bothPerfect = powerLevel >= maxLevel && (partnerCurrentlyMax || partnerRecentlyReleasedMax);

        // Record my max-level release so the partner can answer within the window
        if (powerLevel >= maxLevel) {
            if (bothPerfect) {
                pendingPerfect.remove(target.getUuid());
                pendingPerfect.remove(id);
            } else {
                pendingPerfect.put(id, new PendingPerfect(target.getUuid(), powerLevel, now));
            }
        }

        if (bothPerfect) {
            // consume partner's charge (if still charging)
            chargeStart.remove(target.getUuid());
            chargeLevels.remove(target.getUuid());
            chargeTarget.remove(target.getUuid());
            executePerfectDap(player, target);
            Lang.chargeRelease(player, powerLevel);
            if (partnerLevel != null) Lang.chargeRelease(target, partnerLevel);
            InteractionCooldowns.apply(id, "dap", ModConfig.get().dapCooldownMs);
            InteractionCooldowns.apply(target.getUuid(), "dap", ModConfig.get().dapCooldownMs);
            return;
        }

        executeDap(player, target, powerLevel + 1);
        Lang.chargeRelease(player, powerLevel);
        InteractionCooldowns.apply(id, "dap", ModConfig.get().dapCooldownMs);
        InteractionCooldowns.apply(target.getUuid(), "dap", ModConfig.get().dapCooldownMs);
    }

    public static void cleanup(UUID playerId) {
        pendingHighFive.remove(playerId);
        pendingPerfect.remove(playerId);
        pendingPerfect.entrySet().removeIf(e -> e.getValue().targetId().equals(playerId));
        chargeStart.remove(playerId);
        chargeLevels.remove(playerId);
        chargeTarget.remove(playerId);
        Iterator<Map.Entry<UUID, UUID>> it = chargeTarget.entrySet().iterator();
        while (it.hasNext()) {
            Map.Entry<UUID, UUID> e = it.next();
            if (e.getValue().equals(playerId)) {
                chargeStart.remove(e.getKey());
                chargeLevels.remove(e.getKey());
                it.remove();
            }
        }
        speedHistory.remove(playerId);
        pendingEffects.remove(playerId);
    }

    public static void purgeExpiredCooldowns() {
        long now = TickClock.now();
        pendingHighFive.entrySet().removeIf(e -> now - e.getValue() > HIGH_FIVE_WINDOW_TICKS * 2);
        pendingPerfect.entrySet().removeIf(e -> now - e.getValue().releaseTick() > PERFECT_WINDOW_TICKS * 2);
        int mode = ModConfig.get().chargedDapMode;
        long timeout = (mode == 0) ? CHARGE_TIMEOUT_TICKS : CYCLE_CHARGE_TIMEOUT_TICKS;
        chargeStart.entrySet().removeIf(e -> {
            if (now - e.getValue() > timeout) {
                chargeLevels.remove(e.getKey());
                chargeTarget.remove(e.getKey());
                return true;
            }
            return false;
        });
    }

    private static void executeDap(ServerPlayerEntity sp, ServerPlayerEntity target, int power) {
        if (sp == null || target == null) return;
        ServerWorld w = (ServerWorld) sp.getEntityWorld();
        if (w == null) return;
        DapSession session = DapSessionManager.createSession(
                sp.getUuid(), target.getUuid(), 1.2, DapSession.DapType.NORMAL_DAP);
        Runnable effects = () -> {
            Vec3d mid = sp.getEntityPos().add(target.getEntityPos()).multiply(0.5).add(0, 1.2, 0);
            int particleCount = 4 + power * 2;
            w.spawnParticles(ParticleTypes.CRIT, mid.x, mid.y, mid.z, particleCount, 0.2, 0.2, 0.2, 0.1);
            w.spawnParticles(ParticleTypes.ENCHANTED_HIT, mid.x, mid.y, mid.z, power, 0.15, 0.15, 0.15, 0.08);
            float pitch = 1.0f + power * 0.1f;
            // Single clap sound for dap (not a 2-second loop)
            w.playSound(null, mid.x, mid.y, mid.z, SoundEvents.BLOCK_NOTE_BLOCK_HAT.value(), SoundCategory.PLAYERS, 0.9f, pitch);
            sp.swingHand(net.minecraft.util.Hand.MAIN_HAND, true);
            Lang.dapSuccess(sp, target);
            com.dap.dapurhomies.compat.DanseCompat.playDap(sp, target);
        };
        if (session != null) session.onComplete(effects);
        else effects.run();
    }

    private static void executeHighFive(ServerPlayerEntity p1, ServerPlayerEntity p2) {
        if (p1 == null || p2 == null) return;
        ServerWorld w = (ServerWorld) p1.getEntityWorld();
        if (w == null) return;
        double speed1 = getMaxRecentSpeed(p1.getUuid());
        double speed2 = getMaxRecentSpeed(p2.getUuid());
        double maxSpeed = Math.max(speed1, speed2);
        int tier = 0;
        if (maxSpeed >= SPEED_TIER_3) tier = 3;
        else if (maxSpeed >= SPEED_TIER_2) tier = 2;
        else if (maxSpeed >= SPEED_TIER_1) tier = 1;
        DapSession session = DapSessionManager.createSession(
                p1.getUuid(), p2.getUuid(), 1.0, DapSession.DapType.NORMAL_DAP);
        Vec3d highFivePos = p1.getEntityPos().add(p2.getEntityPos()).multiply(0.5).add(0, 1.5, 0);
        speedHistory.remove(p1.getUuid());
        speedHistory.remove(p2.getUuid());
        final ServerPlayerEntity fp1 = p1, fp2 = p2;
        final Vec3d fPos = highFivePos;
        final int fTier = tier;
        Runnable onComplete = () -> {
            pendingEffects.put(fp1.getUuid(), new PendingHighFive(fp1, fp2, fPos, fTier, TickClock.now() + HIT_EFFECT_DELAY_TICKS));
            fp1.swingHand(net.minecraft.util.Hand.MAIN_HAND, true);
            fp2.swingHand(net.minecraft.util.Hand.MAIN_HAND, true);
            Lang.highFiveSuccess(fp1, fp2, fTier);
            // Perfect high-five (tier 3) plays Danse dap animation
            if (fTier >= 3) {
                com.dap.dapurhomies.compat.DanseCompat.playDap(fp1, fp2);
            } else {
                com.dap.dapurhomies.compat.DanseCompat.playHighFive(fp1, fp2);
            }
            // High-five has its own tier-based sound effects in executeHighFiveEffects(),
            // no need for a 2-second clap loop here
        };
        if (session != null) session.onComplete(onComplete);
        else onComplete.run();
    }

    // Perfect Dap (v0.6.0): dedicated tier-3-level celebration instead of a plain power-6 dap
    private static void executePerfectDap(ServerPlayerEntity p1, ServerPlayerEntity p2) {
        if (p1 == null || p2 == null) return;
        ServerWorld w = (ServerWorld) p1.getEntityWorld();

package com.dap.dapurhomies.interaction;

import net.minecraft.server.MinecraftServer;
import net.minecraft.server.network.ServerPlayerEntity;
import net.minecraft.server.world.ServerWorld;
import net.minecraft.util.math.BlockPos;
import net.minecraft.util.math.Vec3d;

import java.util.UUID;

/**
 * DapSession - ported from original Dap-ur-homie project.
 * Manages a two-player dap session where both players are smoothly moved
 * to face each other at a specific distance, creating a cinematic interaction.
 */
public class DapSession {

    private final UUID playerAId;
    private final UUID playerBId;
    private Vec3d targetA;
    private Vec3d targetB;

    private int tickCount;
    private boolean positioningComplete;

    private final double targetDistance;
    private final double lerpSpeed;
    private final DapType type;

    private Runnable onPositioningComplete;

    public enum DapType {
        NORMAL_DAP,
        PERFECT_DAP,
        FIRE_DAP,
        FIRE_COMBO,
        DAP_HOLD
    }

    public DapSession(UUID playerA, UUID playerB, double targetDistance, DapType type) {
        this.playerAId = playerA;
        this.playerBId = playerB;
        this.targetDistance = targetDistance;
        this.type = type;
        this.tickCount = 0;
        this.positioningComplete = false;

        if (type == DapType.PERFECT_DAP) {
            this.lerpSpeed = 0.95;
        } else {
            this.lerpSpeed = 0.75;
        }
    }

    public void onComplete(Runnable callback) {
        this.onPositioningComplete = callback;
    }

    public void tick(MinecraftServer server) {
        ServerPlayerEntity playerA = server.getPlayerManager().getPlayer(playerAId);
        ServerPlayerEntity playerB = server.getPlayerManager().getPlayer(playerBId);

        // Safety checks - if either player is gone, force complete and end session
        if (playerA == null || playerB == null || playerA.isRemoved() || playerB.isRemoved()) {
            forceComplete(playerA, playerB);
            return;
        }

        // Timeout safety (5 seconds = 100 ticks)
        if (tickCount > 100) {
            forceComplete(playerA, playerB);
            return;
        }

        freezePlayers(playerA, playerB);
        computeTargets(playerA, playerB);
        smoothMoveToTargets(playerA, playerB);
        makeFaceEachOther(playerA, playerB);
        playerA.swingHand(net.minecraft.util.Hand.MAIN_HAND, true);
        playerB.swingHand(net.minecraft.util.Hand.MAIN_HAND, true);

        if (!positioningComplete) {
            checkPositioningComplete(playerA, playerB);
        }

        tickCount++;
    }

    private void freezePlayers(ServerPlayerEntity playerA, ServerPlayerEntity playerB) {
        playerA.setVelocity(Vec3d.ZERO);
        playerB.setVelocity(Vec3d.ZERO);
        playerA.knockedBack = true; // sync velocity to client
        playerB.knockedBack = true;
        playerA.fallDistance = 0;
        playerB.fallDistance = 0;
    }

    private void computeTargets(ServerPlayerEntity playerA, ServerPlayerEntity playerB) {
        Vec3d posA = playerA.getEntityPos();
        Vec3d posB = playerB.getEntityPos();

        Vec3d midpoint = posA.add(posB).multiply(0.5);

        Vec3d direction = posB.subtract(posA);
        if (direction.lengthSquared() < 1e-6) {
            direction = new Vec3d(1, 0, 0);
        }
        direction = direction.normalize();

        double halfDistance = targetDistance / 2.0;
        targetA = midpoint.subtract(direction.multiply(halfDistance));
        targetB = midpoint.add(direction.multiply(halfDistance));

        double targetY = Math.max(posA.y, posB.y);

        ServerWorld world = (ServerWorld) playerA.getEntityWorld();
        BlockPos groundPos = BlockPos.ofFloored(midpoint.x, targetY - 1, midpoint.z);
        if (world.getBlockState(groundPos).isAir()) {
            targetY = Math.min(posA.y, posB.y);
        }

        targetA = new Vec3d(targetA.x, targetY, targetA.z);
        targetB = new Vec3d(targetB.x, targetY, targetB.z);
    }

    private void smoothMoveToTargets(ServerPlayerEntity playerA, ServerPlayerEntity playerB) {
        Vec3d currentA = playerA.getEntityPos();
        Vec3d currentB = playerB.getEntityPos();

        Vec3d newPosA = new Vec3d(
                lerp(currentA.x, targetA.x, lerpSpeed),
                lerp(currentA.y, targetA.y, lerpSpeed),
                lerp(currentA.z, targetA.z, lerpSpeed)
        );

        Vec3d newPosB = new Vec3d(
                lerp(currentB.x, targetB.x, lerpSpeed),
                lerp(currentB.y, targetB.y, lerpSpeed),
                lerp(currentB.z, targetB.z, lerpSpeed)
        );

        playerA.teleport((ServerWorld) playerA.getEntityWorld(), newPosA.x, newPosA.y, newPosA.z,
                java.util.Set.of(), playerA.getYaw(), playerA.getPitch(), false);
        playerB.teleport((ServerWorld) playerB.getEntityWorld(), newPosB.x, newPosB.y, newPosB.z,
                java.util.Set.of(), playerB.getYaw(), playerB.getPitch(), false);
    }

    private void makeFaceEachOther(ServerPlayerEntity playerA, ServerPlayerEntity playerB) {
        Vec3d posA = playerA.getEntityPos();
        Vec3d posB = playerB.getEntityPos();

        double dx = posB.x - posA.x;
        double dz = posB.z - posA.z;
        float yawA = (float) (Math.atan2(dz, dx) * 180 / Math.PI) - 90;
        float yawB = yawA + 180;

        playerA.setYaw(yawA);
        playerA.setBodyYaw(yawA);
        playerA.setHeadYaw(yawA);
        playerA.lastYaw = yawA;
        playerA.lastBodyYaw = yawA;
        playerA.lastHeadYaw = yawA;

        playerB.setYaw(yawB);
        playerB.setBodyYaw(yawB);
        playerB.setHeadYaw(yawB);
        playerB.lastYaw = yawB;
        playerB.lastBodyYaw = yawB;
        playerB.lastHeadYaw = yawB;

        playerA.teleport((ServerWorld) playerA.getEntityWorld(), posA.x, posA.y, posA.z,
                java.util.Set.of(), yawA, playerA.getPitch(), false);
        playerB.teleport((ServerWorld) playerB.getEntityWorld(), posB.x, posB.y, posB.z,
                java.util.Set.of(), yawB, playerB.getPitch(), false);
    }

    private void checkPositioningComplete(ServerPlayerEntity playerA, ServerPlayerEntity playerB) {
        double distA = playerA.getEntityPos().distanceTo(targetA);
        double distB = playerB.getEntityPos().distanceTo(targetB);

        double threshold = (type == DapType.PERFECT_DAP) ? 0.35 : 0.25;
        if (distA < threshold && distB < threshold) {
            positioningComplete = true;

            if (onPositioningComplete != null) {
                onPositioningComplete.run();
            }

            this.tickCount = 99; // Force near-timeout so session ends next tick
        }
    }

    private void forceComplete(ServerPlayerEntity playerA, ServerPlayerEntity playerB) {
        if (!positioningComplete) {
            positioningComplete = true;
            if (onPositioningComplete != null) {
                onPositioningComplete.run();
            }
        }
    }

    private double lerp(double current, double target, double factor) {
        return current + (target - current) * factor;
    }

    public UUID getPlayerAId() { return playerAId; }
    public UUID getPlayerBId() { return playerBId; }
    public boolean isPositioningComplete() { return positioningComplete; }
    public int getTickCount() { return tickCount; }
}

package com.dap.dapurhomies.interaction;

import net.minecraft.server.MinecraftServer;

import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.Set;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

/**
 * DapSessionManager - ported from original Dap-ur-homie project.
 * Manages active dap sessions between pairs of players.
 */
public class DapSessionManager {
    private static final Map<UUID, DapSession> activeSessions = new ConcurrentHashMap<>();
    private static final Set<UUID> playersInSession = ConcurrentHashMap.newKeySet();

    public static DapSession createSession(UUID playerA, UUID playerB, double targetDistance, DapSession.DapType type) {
        if (isInSession(playerA) || isInSession(playerB)) {
            return null;
        }
        DapSession session = new DapSession(playerA, playerB, targetDistance, type);
        activeSessions.put(playerA, session);
        playersInSession.add(playerA);
        playersInSession.add(playerB);
        return session;
    }

    public static DapSession getSession(UUID playerId) {
        DapSession session = activeSessions.get(playerId);
        if (session != null) return session;
        for (DapSession s : activeSessions.values()) {
            if (s.getPlayerBId().equals(playerId)) {
                return s;
            }
        }
        return null;
    }

    public static boolean isInSession(UUID playerId) {
        return playersInSession.contains(playerId);
    }

    public static void removeSession(UUID playerA) {
        DapSession session = activeSessions.remove(playerA);
        if (session != null) {
            playersInSession.remove(session.getPlayerAId());
            playersInSession.remove(session.getPlayerBId());
        }
    }

    public static void removeSessionForPlayer(UUID playerId) {
        DapSession session = getSession(playerId);
        if (session != null) {
            removeSession(session.getPlayerAId());
        }
    }

    public static void tick(MinecraftServer server) {
        List<DapSession> sessionsToTick = new ArrayList<>(activeSessions.values());
        for (DapSession session : sessionsToTick) {
            session.tick(server);
            if (session.isPositioningComplete() || session.getTickCount() > 100) {
                UUID playerA = session.getPlayerAId();
                UUID playerB = session.getPlayerBId();
                activeSessions.remove(playerA);
                playersInSession.remove(playerA);
                playersInSession.remove(playerB);
            }
        }
    }

    public static void clearAll() {
        activeSessions.clear();
        playersInSession.clear();
    }

    public static void cleanup(UUID playerId) {
        removeSessionForPlayer(playerId);
    }
}

/**
 * Fall Catch system - ported from original Dap-ur-homie project.
 * Players can enter "catch ready" mode to catch falling players within range.
 * Pure server-side implementation.
 */
public class FallCatchHandler {

    public static final double CATCH_RANGE_HORIZONTAL = 3.0;
    public static final double CATCH_RANGE_VERTICAL = 4.0;
    public static final long CATCH_WINDOW_TICKS = TickClock.msToTicks(500);
    public static final long CATCH_COOLDOWN_TICKS = TickClock.msToTicks(1000);

    // ===== STATE MAPS =====
    private static final Map<UUID, Long> catchReadyTime = new HashMap<>();   // playerId -> readyStartTime
    private static final Map<UUID, Long> catchCooldowns = new HashMap<>();     // playerId -> cooldownEndTime
    private static final Map<UUID, Boolean> successfulCatch = new HashMap<>(); // playerId -> caught someone this window

    public static void register() {
        // Tick is driven by InteractionModules.tickAll(); do not register another server tick here.
        ServerLivingEntityEvents.ALLOW_DAMAGE.register((entity, source, amount) -> {
            if (!(entity instanceof ServerPlayerEntity fallingPlayer)) return true;
            if (DapUrHomiesMod.isModEnabled() == false) return true;
            if (!source.isOf(DamageTypes.FALL)) return true;
            if (fallingPlayer.fallDistance < 3.0f) return true;
            if (!ModConfig.get().enableGrabYeet) return true; // Use grab toggle for now

            ServerPlayerEntity catcher = findCatcher(fallingPlayer);
            if (catcher != null) {
                playCatchEffects(fallingPlayer, catcher);
                successfulCatch.put(catcher.getUuid(), true);
                return false; // Cancel fall damage
            }
            return true;
        });
    }

    // ===== PUBLIC API =====

    public static ActionResult enterCatchReady(ServerPlayerEntity player) {
        if (player == null) return ActionResult.PASS;
        if (!ModConfig.get().enableGrabYeet) return ActionResult.PASS;
        if (!PlayerPermissionManager.canInitiate(player, "grab")) {
            Lang.interactionBlocked(player, "Grab is disabled for you!", "你的抓取功能已关闭！");
            return ActionResult.FAIL;
        }

        UUID playerId = player.getUuid();

        if (isOnCatchCooldown(playerId)) {
            Lang.sendActionBar(player, "§c Catch on cooldown!", "§c 接住冷却中！");
            return ActionResult.FAIL;
        }

        if (catchReadyTime.containsKey(playerId)) {
            // Already ready, cancel it
            catchReadyTime.remove(playerId);
            successfulCatch.remove(playerId);
            Lang.sendActionBar(player, "§7 Catch ready cancelled.", "§7 取消接住准备。");
            return ActionResult.SUCCESS;
        }

        catchReadyTime.put(playerId, TickClock.now());
        successfulCatch.put(playerId, false);
        Lang.sendActionBar(player, "§a Catch ready! (500ms window)", "§a 准备接住！(500ms窗口)");
        return ActionResult.SUCCESS;
    }

    public static boolean isOnCatchCooldown(UUID playerId) {
        Long cooldownEnd = catchCooldowns.get(playerId);
        if (cooldownEnd == null) return false;
        if (TickClock.now() >= cooldownEnd) {
            catchCooldowns.remove(playerId);
            return false;
        }
        return true;
    }

    // ===== TICK =====

    public static void tick(MinecraftServer server) {
        long now = TickClock.now();

        Iterator<Map.Entry<UUID, Long>> it = catchReadyTime.entrySet().iterator();
        while (it.hasNext()) {
            Map.Entry<UUID, Long> entry = it.next();
            UUID playerId = entry.getKey();
            long readyTime = entry.getValue();

            // Window expired
            if (now - readyTime > CATCH_WINDOW_TICKS) {
                if (!successfulCatch.getOrDefault(playerId, false)) {
                    // Missed - apply cooldown
                    catchCooldowns.put(playerId, now + CATCH_COOLDOWN_TICKS);

                    ServerPlayerEntity player = server.getPlayerManager().getPlayer(playerId);
                    if (player != null) {
                        Lang.sendActionBar(player, "§c Catch missed! 1s cooldown", "§c 接住失败！1秒冷却");
                    }
                }
                it.remove();
                successfulCatch.remove(playerId);
            }
        }
    }

    // ===== FIND CATCHER =====

    private static ServerPlayerEntity findCatcher(ServerPlayerEntity fallingPlayer) {
        ServerWorld world = fallingPlayer.getEntityWorld();
        long now = TickClock.now();

        Box searchBox = new Box(
                fallingPlayer.getX() - CATCH_RANGE_HORIZONTAL,
                fallingPlayer.getY() - CATCH_RANGE_VERTICAL,
                fallingPlayer.getZ() - CATCH_RANGE_HORIZONTAL,
                fallingPlayer.getX() + CATCH_RANGE_HORIZONTAL,
                fallingPlayer.getY() + CATCH_RANGE_VERTICAL,
                fallingPlayer.getZ() + CATCH_RANGE_HORIZONTAL
        );

        for (PlayerEntity player : world.getPlayers()) {
            if (player == fallingPlayer) continue;
            if (!(player instanceof ServerPlayerEntity catcher)) continue;

            UUID catcherId = catcher.getUuid();

            // Must be in catch ready mode
            Long readyTime = catchReadyTime.get(catcherId);
            if (readyTime == null) continue;

            // Check window
            if (now - readyTime > CATCH_WINDOW_TICKS) continue;

            // Check cooldown
            if (isOnCatchCooldown(catcherId)) continue;

            // Check range
            if (!searchBox.contains(catcher.getEntityPos())) continue;

            // Permissions: catcher must be allowed to grab, falling player must accept grabs
            if (!PlayerPermissionManager.canInitiate(catcher, "grab")) continue;
            if (!PlayerPermissionManager.hasPermission(catcher, "dap-ur-homies.admin")
                    && !PlayerPermissionManager.canReceive(fallingPlayer, "grab")) continue;

            double dx = fallingPlayer.getX() - catcher.getX();
            double dz = fallingPlayer.getZ() - catcher.getZ();
            double horizontalDist = Math.sqrt(dx * dx + dz * dz);

            if (horizontalDist <= CATCH_RANGE_HORIZONTAL) {
                return catcher;
            }
        }

        return null;
    }

    public static ActionResult tryGrab(ServerPlayerEntity holder, ServerPlayerEntity target) {
        if (holder == null || target == null) return ActionResult.PASS;
        if (holder.equals(target)) return ActionResult.PASS;
        if (!ModConfig.get().enableGrabYeet) return ActionResult.PASS;
        if (holder.getEntityWorld() != target.getEntityWorld()) return ActionResult.PASS;

        UUID hid = holder.getUuid();
        UUID tid = target.getUuid();

        // Already holding or being held
        if (holding.containsKey(hid) || heldBy.containsKey(hid)) {
            Lang.sendActionBar(holder, "\u00a7cYou are already holding someone!");
            return ActionResult.FAIL;
        }
        if (holding.containsKey(tid) || heldBy.containsKey(tid)) {
            Lang.sendActionBar(holder, "\u00a7cTarget is already in a grab!");
            return ActionResult.FAIL;
        }

        // Distance check
        if (holder.distanceTo(target) > ModConfig.get().grabRange) {
            Lang.sendActionBar(holder, "\u00a7cToo far to grab!");
            return ActionResult.FAIL;
        }

        // Throw immunity check (1 second after being thrown)
        if (ThrowTracker.isImmune(tid)) {
            Lang.sendActionBar(holder, "\u00a7cTarget is recovering from being thrown!");
            return ActionResult.FAIL;
        }

        // Permission check
        if (!PlayerPermissionManager.canInitiate(holder, "grab")) {
            Lang.interactionBlocked(holder, "Grab is disabled for you!", "\u4f60\u7684\u6293\u53d6\u529f\u80fd\u5df2\u5173\u95ed\uff01");
            return ActionResult.FAIL;
        }
        if (!PlayerPermissionManager.hasPermission(holder, "dap-ur-homies.admin")
                && !PlayerPermissionManager.canReceive(target, "grab")) {
            Lang.targetDisabledType(holder, target, "grab");
            return ActionResult.FAIL;
        }

        // Protection check
        if (!PlayerPermissionManager.hasPermission(holder, "dap-ur-homies.admin")
                && !ProtectionCompat.canInteract(holder, target)) {
            Lang.protectedTarget(holder, target);
            return ActionResult.FAIL;
        }

        // Cooldown check
        if (InteractionCooldowns.rejectIfBlocked(holder, "grab")) return ActionResult.FAIL;

        // Apply grab
        holding.put(hid, tid);
        heldBy.put(tid, hid);
        lastGrabTick.put(hid, TickClock.now());   // 记录抓取时刻，防止同一左键被误判为放下

        // No riding. tickCarryPositions positions the held player overhead each tick.
        // (Riding can't move a no-mod client's camera above the head; teleport can.)

        // Effects
        ServerWorld world = holder.getEntityWorld();
        Vec3d pos = target.getEntityPos();
        world.playSound(null, pos.x, pos.y, pos.z,
                SoundEvents.ENTITY_PLAYER_ATTACK_STRONG, SoundCategory.PLAYERS, 1.0f, 0.8f);
        world.spawnParticles(ParticleTypes.CLOUD, pos.x, pos.y + 0.5, pos.z, 5, 0.2, 0.2, 0.2, 0.02);

        Lang.grabSuccess(holder, target);
        InteractionCooldowns.apply(hid, "grab", ModConfig.get().grabCooldownMs);
        com.dap.dapurhomies.compat.DanseCompat.playGrab(holder, target); // Danse animation

        return ActionResult.SUCCESS;
    }

    public static ActionResult yeetCurrentHold(ServerPlayerEntity holder, float power) {
        if (holder == null) return ActionResult.PASS;
        UUID hid = holder.getUuid();
        UUID tid = holding.get(hid);
        if (tid == null) return ActionResult.PASS;

        var server = holder.getEntityWorld().getServer();
        if (server == null) {
            cleanup(hid);
            return ActionResult.FAIL;
        }
        ServerPlayerEntity target = server.getPlayerManager().getPlayer(tid);
        if (target == null) {
            cleanup(hid);
            return ActionResult.FAIL;
        }

        // Hunger check (from original project)
        if (!holder.isCreative()) {
            if (holder.getHungerManager().getFoodLevel() < 6) {
                Lang.sendActionBar(holder, "Too hungry to throw!", "太饿了，无法投掷！");
                return ActionResult.FAIL;
            }
            holder.getHungerManager().addExhaustion(18.0f);
        }

        // Calculate throw direction
        Vec3d holderLook = holder.getRotationVec(1.0f);
        Vec3d throwDir = new Vec3d(holderLook.x, 0, holderLook.z).normalize();
        if (throwDir.lengthSquared() < 0.001) throwDir = new Vec3d(0, 0, 1);

        // Calculate power based on charge (0.0 - 1.0)
        double scaledPower = MIN_THROW_POWER + (MAX_THROW_POWER - MIN_THROW_POWER) * Math.max(0.0f, Math.min(1.0f, power));

        // Calculate velocity
        double horizontalSpeed = 1.3 * scaledPower;
        double verticalSpeed = Math.min(0.8 + scaledPower * 0.15, 1.5);

        Vec3d velocity = new Vec3d(
                throwDir.x * horizontalSpeed,
                verticalSpeed,
                throwDir.z * horizontalSpeed
        );

        // Remove passenger relationship
        target.stopRiding();

        // Broadcast passenger-clear so every client immediately drops the riding visual.
        var syncServer2 = holder.getEntityWorld().getServer();
        if (syncServer2 != null) {
            var clearPkt = new net.minecraft.network.packet.s2c.play.EntityPassengersSetS2CPacket(holder);
            for (ServerPlayerEntity p : syncServer2.getPlayerManager().getPlayerList()) {
                p.networkHandler.sendPacket(clearPkt);
            }
        }

        // Lock held player's rotation to holder's rotation for 1 tick (view correction)
        float throwYaw = holder.getYaw();
        float throwPitch = holder.getPitch();
        target.setYaw(throwYaw);
        target.setBodyYaw(throwYaw);
        target.setHeadYaw(throwYaw);
        target.setPitch(throwPitch);
        target.lastYaw = throwYaw;
        target.lastBodyYaw = throwYaw;
        target.lastHeadYaw = throwYaw;
        target.lastPitch = throwPitch;

        // Force teleport to sync rotation immediately
        Vec3d releasePos = holder.getEntityPos().add(holder.getRotationVec(1.0f).multiply(1.5).multiply(1, 0, 1)).add(0, 0.5, 0);
        target.refreshPositionAndAngles(releasePos.x, releasePos.y, releasePos.z, throwYaw, throwPitch);
        target.networkHandler.requestTeleport(releasePos.x, releasePos.y, releasePos.z, throwYaw, throwPitch);

        // Clear shield mode if active
        if (HumanShieldController.isActive(hid)) {
            HumanShieldController.disableShieldMode(holder, false);
        }

        // Remove from holding maps
        holding.remove(hid);
        heldBy.remove(tid);

        // Remove charge BossBar
        ServerBossBar throwBar = chargeBossBars.remove(hid);
        if (throwBar != null) {
            throwBar.removePlayer(holder);
        }

        // Add to pending throws (delayed velocity application)
        ThrowTracker.schedulePendingThrow(tid, velocity);

        // Track as thrown player
        boolean wasOnFire = target.isOnFire();
        ThrowTracker.registerThrown(tid, wasOnFire);

        // Mark for trail particle effects
        LaunchedPlayerTracker.markPlayerAsLaunched(tid);

        // Apply throw immunity after release.
        ThrowTracker.applyImmunity(tid);

        // Effects
        ServerWorld world = holder.getEntityWorld();
        Vec3d pos = target.getEntityPos();
        world.playSound(null, pos.x, pos.y, pos.z,
                SoundEvents.ENTITY_PLAYER_ATTACK_SWEEP, SoundCategory.PLAYERS, 1.2f, 0.9f);
        world.spawnParticles(ParticleTypes.CLOUD, pos.x, pos.y + 0.5, pos.z, 8, 0.3, 0.3, 0.3, 0.05);

        Lang.yeetSuccess(holder, target);
        InteractionCooldowns.apply(hid, "grab", ModConfig.get().grabCooldownMs);
        com.dap.dapurhomies.compat.DanseCompat.playYeet(target); // Danse animation
        lastGrabTick.remove(hid);
        return ActionResult.SUCCESS;
    }

    public static ActionResult startCharge(ServerPlayerEntity holder) {
        if (holder == null) return ActionResult.PASS;
        if (!isHolding(holder)) return ActionResult.PASS;

        UUID hid = holder.getUuid();

        // Already charging - ignore, left-click handles release
        if (chargeStartTime.containsKey(hid)) {
            Lang.sendActionBar(holder, "\u00a7eAlready charging! Left-click to throw!",
                "\u00a7e\u5df2\u5728\u84c4\u529b\u4e2d\uff01\u5de6\u952e\u6295\u51fa\uff01");
            return ActionResult.SUCCESS;
        }

        // Start charging
        chargeStartTime.put(hid, TickClock.now());

        // Create BossBar for charge progress (like Ender Dragon health bar)
        ServerBossBar bossBar = new ServerBossBar(
                com.dap.dapurhomies.Lang.t(holder, "\u26a1 Charging Throw \u26a1", "\u26a1 \u84c4\u529b\u6295\u63b7\u4e2d \u26a1"),
                BossBar.Color.YELLOW,
                BossBar.Style.PROGRESS);
        bossBar.setPercent(0.0f);
        bossBar.addPlayer(holder);
        chargeBossBars.put(hid, bossBar);

        Lang.sendActionBar(holder, "\u00a7e\u26a1 Charge started! Left-click to throw!",
            "\u00a7e\u26a1 \u84c4\u529b\u5f00\u59cb\uff01\u5de6\u952e\u6295\u51fa\uff01");
        return ActionResult.SUCCESS;
    }

    public static ActionResult releaseCharge(ServerPlayerEntity holder) {
        if (holder == null) return ActionResult.PASS;
        UUID hid = holder.getUuid();

        // Remove BossBar
        ServerBossBar bossBar = chargeBossBars.remove(hid);
        if (bossBar != null) {
            bossBar.removePlayer(holder);
        }

        Long start = chargeStartTime.remove(hid);
        if (start == null) {
            // Not charging - do instant throw with default power
            return yeetCurrentHold(holder, 0.5f);
        }

        long elapsed = TickClock.now() - start;
        float power = Math.min(1.0f, (float) elapsed / CHARGE_MAX_TICKS);

        return yeetCurrentHold(holder, power);
    }

    private static void tickCharge(MinecraftServer server) {
        long now = TickClock.now();
        Iterator<Map.Entry<UUID, Long>> it = chargeStartTime.entrySet().iterator();
        while (it.hasNext()) {
            Map.Entry<UUID, Long> entry = it.next();
            UUID hid = entry.getKey();
            long start = entry.getValue();

            ServerPlayerEntity holder = server.getPlayerManager().getPlayer(hid);
            if (holder == null || !isHolding(holder)) {
                // Remove any BossBar left behind by an invalid charge.
                ServerBossBar bar = chargeBossBars.remove(hid);
                if (bar != null && holder != null) bar.removePlayer(holder);
                it.remove();
                continue;
            }

            // Auto-release at full charge inline because releaseCharge() would mutate
            // chargeStartTime while it is being iterated.
            if (now - start >= CHARGE_AUTO_RELEASE_TICKS) {
                ServerBossBar bar = chargeBossBars.remove(hid);
                if (bar != null) bar.removePlayer(holder);
                it.remove(); // safe removal via iterator
                yeetCurrentHold(holder, 1.0f);
                continue;
            }

            // Charge particles
            long elapsed = now - start;
            float progress = Math.min(1.0f, (float) elapsed / CHARGE_MAX_TICKS);
            if (server.getOverworld().getTime() % 5 == 0) {
                Vec3d pos = holder.getEntityPos().add(0, 1, 0);
                ServerWorld world = holder.getEntityWorld();
                world.spawnParticles(ParticleTypes.CRIT, pos.x, pos.y, pos.z,
                    1 + (int)(progress * 3), 0.2, 0.2, 0.2, 0.05);
            }

            // Update BossBar progress (like Ender Dragon health bar)
            ServerBossBar bossBar = chargeBossBars.get(hid);
            if (bossBar != null) {
                bossBar.setPercent(progress);
                // Change color based on progress
                if (progress >= 1.0f) {
                    bossBar.setColor(BossBar.Color.RED);
                    bossBar.setName(com.dap.dapurhomies.Lang.t(holder,
                        "\u26a1 MAX CHARGE - RELEASE! \u26a1",
                        "\u26a1 \u84c4\u529b\u5df2\u6ee1 - \u91ca\u653e! \u26a1"));
                } else if (progress >= 0.7f) {
                    bossBar.setColor(BossBar.Color.PINK);
                } else if (progress >= 0.4f) {
                    bossBar.setColor(BossBar.Color.GREEN);
                }
            }
        }
    }

    public static ActionResult tryEscape(ServerPlayerEntity held) {
        if (held == null) return ActionResult.PASS;
        UUID tid = held.getUuid();
        UUID hid = heldBy.get(tid);
        if (hid == null) return ActionResult.PASS;

        var server = held.getEntityWorld().getServer();
        if (server == null) {
            cleanup(tid);
            return ActionResult.FAIL;
        }
        ServerPlayerEntity holder = server.getPlayerManager().getPlayer(hid);
        if (holder == null) {
            cleanup(tid);
            return ActionResult.FAIL;
        }

        // 20% chance to escape
        if (held.getRandom().nextFloat() < 0.2f) {
            held.stopRiding();
            holding.remove(hid);
            heldBy.remove(tid);
            chargeStartTime.remove(hid);

            // Remove charge BossBar
            if (holder != null) {
                ServerBossBar escapeBar = chargeBossBars.remove(hid);
                if (escapeBar != null) {
                    escapeBar.removePlayer(holder);
                }
            }

            // Clear shield mode if active
            if (HumanShieldController.isActive(hid)) {
                HumanShieldController.clearForEscape(hid);
            }

            // Push both apart
            Vec3d dir = held.getEntityPos().subtract(holder.getEntityPos()).normalize();
            if (dir.lengthSquared() < 0.001) dir = new Vec3d(1, 0, 0);
            held.setVelocity(dir.x * 0.5, 0.3, dir.z * 0.5);
            holder.setVelocity(-dir.x * 0.3, 0.1, -dir.z * 0.3);
            held.knockedBack = true;
            holder.knockedBack = true;

            Lang.escapeSuccess(held);
            return ActionResult.SUCCESS;
        } else {
            Lang.escapeFail(held);
            return ActionResult.FAIL;
        }
    }

    private static void tickCarryPositions(MinecraftServer server) {
        carryTickCounter++;
        boolean doTeleport = (carryTickCounter % 2 == 0);   // every 2 ticks

        for (var e : holding.entrySet()) {
            UUID hid = e.getKey();
            ServerPlayerEntity holder = server.getPlayerManager().getPlayer(hid);
            ServerPlayerEntity held = server.getPlayerManager().getPlayer(e.getValue());
            if (holder == null || held == null) continue;
            if (holder.getEntityWorld() != held.getEntityWorld()) {
                cleanup(hid);
                continue;
            }
            if (HumanShieldController.isActive(hid)) {
                // Keep the intermediate armor stand anchored to the holder
                ArmorStandEntity stand = HumanShieldController.getStand(hid);
                if (stand != null && stand.isAlive()) {
                    Vec3d hp = holder.getEntityPos();
                    stand.setPosition(hp.x, hp.y, hp.z);
                    continue;
                }
                // Stand was killed/removed externally: drop shield mode and fall back to direct carry,
                // otherwise the held player stays frozen forever (travel cancelled, never re-teleported).
                HumanShieldController.disableShieldMode(holder, false);
            }

            // Always zero velocity & fall distance so the held player never falls
            held.setVelocity(0, 0, 0);
            held.knockedBack = true; // sync the zeroed velocity to the client
            held.fallDistance = 0;

            if (doTeleport) {
                double x = holder.getX();
                double y = holder.getY() + holder.getHeight() + 0.1;
                double z = holder.getZ();
                // requestTeleport every 2 ticks —— held player never falls, position authoritative.
                // Pass held's OWN yaw/pitch (free look, no camera lock).
                held.refreshPositionAndAngles(x, y, z, held.getYaw(), held.getPitch());
                held.networkHandler.requestTeleport(x, y, z, held.getYaw(), held.getPitch());
            }
        }
    }

    private static void tickHeldPlayers(MinecraftServer server) {
        // ConcurrentHashMap supports removal during iteration
        for (Map.Entry<UUID, UUID> entry : holding.entrySet()) {
            UUID hid = entry.getKey();
            UUID tid = entry.getValue();

            ServerPlayerEntity holder = server.getPlayerManager().getPlayer(hid);
            ServerPlayerEntity held = server.getPlayerManager().getPlayer(tid);

            if (holder == null || held == null || !holder.isAlive() || !held.isAlive()) {
                cleanup(hid);
                continue;
            }

            // Check distance
            if (holder.distanceTo(held) > THROW_RANGE * 2) {
                // Auto-drop when the pair drifts apart; cleanup also removes shield stand/state
                cleanup(hid);
                continue;
            }

            // Freeze holder movement while carrying (user request: cannot move but can look around)
            holder.setVelocity(0, holder.getVelocity().y, 0);
            holder.knockedBack = true;

            // Carrying slowdown is handled by CarryingSlowdown handler via AttributeModifier
        }
    }

    public static void cleanup(UUID playerId) {
        MinecraftServer server = DapUrHomiesMod.getServer();

        // If player is a holder
        UUID heldId = holding.remove(playerId);
        if (heldId != null) {
            heldBy.remove(heldId);
            if (server != null) {
                ServerPlayerEntity held = server.getPlayerManager().getPlayer(heldId);
                if (held != null) {
                    held.stopRiding();
                }
            }
        }

        // Always release this player's own shield stand, even if the holding entry was already gone
        HumanShieldController.cleanup(playerId);

        // If player is being held
        UUID holderId = heldBy.remove(playerId);
        if (holderId != null) {
            holding.remove(holderId);
            if (server != null) {
                ServerPlayerEntity holder = server.getPlayerManager().getPlayer(holderId);
                if (holder != null) {
                    for (Entity passenger : new ArrayList<>(holder.getPassengerList())) {
                        if (passenger instanceof ServerPlayerEntity && passenger.getUuid().equals(playerId)) {
                            passenger.stopRiding();
                        }
                    }
                }
            }
            HumanShieldController.cleanup(holderId);
        }

        ThrowTracker.cleanup(playerId);
        chargeStartTime.remove(playerId);

        // Remove charge BossBar
        ServerBossBar bar = chargeBossBars.remove(playerId);
        if (bar != null) {
            bar.clearPlayers(); // Remove from all viewers
        }

        // Clean up shield swap cooldown
        HumanShieldController.removeSwapCooldown(playerId);

        lastGrabTick.remove(playerId);

package com.dap.dapurhomies.interaction;

import com.dap.dapurhomies.Lang;
import com.dap.dapurhomies.config.ModConfig;
import com.dap.dapurhomies.permission.PlayerPermissionManager;
import com.dap.dapurhomies.util.TickClock;
import net.minecraft.entity.effect.StatusEffectInstance;

import net.minecraft.sound.SoundEvents;

import java.util.HashSet;
import java.util.Iterator;
import java.util.Map;
import java.util.Set;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

public class HugHandler {
    private static final Map<UUID, UUID> hugPairs = new ConcurrentHashMap<>();
    private static final Map<UUID, Long> hugStart = new ConcurrentHashMap<>();
    private static final long HUG_DURATION_TICKS = TickClock.msToTicks(3000);

    public static ActionResult handleHug(ServerPlayerEntity sp, ServerPlayerEntity target) {
        if (sp == null || target == null) return ActionResult.PASS;
        if (!ModConfig.get().enableHug) return ActionResult.PASS;
        if (!sp.isAlive() || !target.isAlive()) return ActionResult.PASS;
        if (sp.getEntityWorld() != target.getEntityWorld()) return ActionResult.PASS;
        if (!PlayerPermissionManager.canInitiate(sp, "hug")) {
            Lang.interactionBlocked(sp, "Hug is disabled for you!", "你的拥抱功能已关闭！");
            return ActionResult.FAIL;
        }
        if (InteractionCooldowns.rejectIfBlocked(sp, "hug")) return ActionResult.FAIL;
        if (!PlayerPermissionManager.hasPermission(sp, "dap-ur-homies.admin")
                && !PlayerPermissionManager.canReceive(target, "hug")) {
            Lang.targetDisabledType(sp, target, "hug");
            return ActionResult.FAIL;
        }
        double dist = sp.distanceTo(target);
        if (dist > ModConfig.get().hugRange) return ActionResult.PASS;
        if (!startHug(sp, target)) return ActionResult.PASS;
        return ActionResult.SUCCESS;
    }

    public static void tick(MinecraftServer server) {
        if (server == null) return;
        long now = TickClock.now();
        Iterator<Map.Entry<UUID, UUID>> it = hugPairs.entrySet().iterator();
        Set<UUID> processed = new HashSet<>();
        while (it.hasNext()) {
            Map.Entry<UUID, UUID> e = it.next();
            if (processed.contains(e.getKey())) continue;
            UUID id1 = e.getKey(), id2 = e.getValue();
            processed.add(id1); processed.add(id2);

            ServerPlayerEntity p1 = server.getPlayerManager().getPlayer(id1);
            ServerPlayerEntity p2 = server.getPlayerManager().getPlayer(id2);
            if (p1 == null || p2 == null || !p1.isAlive() || !p2.isAlive()
                    || p1.getEntityWorld() != p2.getEntityWorld()) {
                if (p1 != null && p2 != null) clearHugEffects(p1, p2);
                it.remove();
                hugPairs.remove(id2);
                hugStart.remove(id1);
                hugStart.remove(id2);
                continue;
            }

            long elapsed = now - hugStart.getOrDefault(id1, now);
            if (elapsed > HUG_DURATION_TICKS) {
                endHug(p1, p2);
                it.remove();
                hugPairs.remove(id2);
                hugStart.remove(id1);
                hugStart.remove(id2);
                processed.add(id2);
                continue;
            }

            // Apply slowness/mining fatigue with longer duration (refresh every 2 seconds instead of every tick)
            // This avoids creating 4 StatusEffectInstance objects per tick per hug pair
            if (now % 40 == 0 || p1.getStatusEffect(StatusEffects.SLOWNESS) == null) {
                p1.addStatusEffect(new StatusEffectInstance(StatusEffects.SLOWNESS, 60, 5, false, false));
                p1.addStatusEffect(new StatusEffectInstance(StatusEffects.MINING_FATIGUE, 60, 4, false, false));
            }
            if (now % 40 == 0 || p2.getStatusEffect(StatusEffects.SLOWNESS) == null) {
                p2.addStatusEffect(new StatusEffectInstance(StatusEffects.SLOWNESS, 60, 5, false, false));
                p2.addStatusEffect(new StatusEffectInstance(StatusEffects.MINING_FATIGUE, 60, 4, false, false));
            }

            if (now % 20 == 0) {
                p1.addStatusEffect(new StatusEffectInstance(StatusEffects.REGENERATION, 60, 0, false, false));
                p2.addStatusEffect(new StatusEffectInstance(StatusEffects.REGENERATION, 60, 0, false, false));
                Vec3d mid = p1.getEntityPos().add(p2.getEntityPos()).multiply(0.5);
                ServerWorld w = p1.getEntityWorld();
                if (w != null) {
                    w.spawnParticles(ParticleTypes.HEART, mid.x, mid.y + 1.2, mid.z, 2, 0.3, 0.3, 0.3, 0);
                    w.playSound(null, mid.x, mid.y, mid.z, SoundEvents.ENTITY_PLAYER_LEVELUP, SoundCategory.PLAYERS, 0.3f, 1.5f);
                }
            }
        }
    }

    public static boolean isHugging(UUID id) {
        return hugPairs.containsKey(id);
    }

    public static void cleanup(UUID id) {
        UUID partner = hugPairs.remove(id);
        if (partner != null) {
            hugPairs.remove(partner);
            hugStart.remove(partner);
        }
        hugStart.remove(id);
    }

    private static boolean startHug(ServerPlayerEntity p1, ServerPlayerEntity p2) {
        if (hugPairs.containsKey(p1.getUuid()) || hugPairs.containsKey(p2.getUuid())) return false;
        long now = TickClock.now();
        hugPairs.put(p1.getUuid(), p2.getUuid());
        hugPairs.put(p2.getUuid(), p1.getUuid());
        hugStart.put(p1.getUuid(), now);
        hugStart.put(p2.getUuid(), now);
        Lang.hugStart(p1, p2);
        com.dap.dapurhomies.compat.DanseCompat.playHug(p1, p2); // Danse animation
        return true;
    }

    /** Remove only the amplified effects this handler applied, leaving other sources untouched. */
    private static void clearHugEffects(ServerPlayerEntity p1, ServerPlayerEntity p2) {
        clearHugEffects(p1);
        clearHugEffects(p2);
    }

    private static void clearHugEffects(ServerPlayerEntity player) {
        StatusEffectInstance slow = player.getStatusEffect(StatusEffects.SLOWNESS);
        if (slow != null && slow.getAmplifier() >= 5) player.removeStatusEffect(StatusEffects.SLOWNESS);
        StatusEffectInstance fatigue = player.getStatusEffect(StatusEffects.MINING_FATIGUE);
        if (fatigue != null && fatigue.getAmplifier() >= 4) player.removeStatusEffect(StatusEffects.MINING_FATIGUE);
    }

    private static void endHug(ServerPlayerEntity p1, ServerPlayerEntity p2) {
        clearHugEffects(p1, p2);
        Lang.hugEnd(p1, p2);
        InteractionCooldowns.applyShared(p1.getUuid(), p2.getUuid(), "hug", ModConfig.get().hugCooldownMs);
    }
}

    private static final Map<UUID, ArmorStandEntity> shieldStands = new HashMap<>(); // holder -> armorstand
    private static final Map<UUID, Boolean> shieldModeActive = new HashMap<>();     // holder -> active
    private static final Map<UUID, Long> shieldSwapCooldown = new HashMap<>();      // holder -> cooldown end time

    private HumanShieldController() {
    }

    static ActionResult toggleShieldMode(ServerPlayerEntity holder) {
        if (holder == null) return ActionResult.PASS;
        UUID hid = holder.getUuid();
        if (GrabYeetHandler.getHeldId(hid) == null) return ActionResult.PASS;

        // Check shield swap cooldown
        Long cooldownEnd = shieldSwapCooldown.get(hid);
        if (cooldownEnd != null && TickClock.now() < cooldownEnd) {
            long remainingTicks = cooldownEnd - TickClock.now();
            long remaining = (remainingTicks + 19) / 20;
            Lang.sendActionBar(holder, "\u00a7c\ud83d\udee1 Shield swap cooldown! " + remaining + "s",
                "\u00a7c\ud83d\udee1 \u76fe\u724c\u5207\u6362\u51b7\u5374\u4e2d\uff01\u8fd8\u5269 " + remaining + "\u79d2");
            return ActionResult.FAIL;
        }

        if (Boolean.TRUE.equals(shieldModeActive.get(hid))) {
            disableShieldMode(holder, true);
        } else {
            enableShieldMode(holder);
        }
        return ActionResult.SUCCESS;
    }

    private static void enableShieldMode(ServerPlayerEntity holder) {
        UUID hid = holder.getUuid();
        UUID tid = GrabYeetHandler.getHeldId(hid);
        if (tid == null) return;

        var server = holder.getEntityWorld().getServer();
        if (server == null) return;
        ServerPlayerEntity held = server.getPlayerManager().getPlayer(tid);
        if (held == null) return;

        // Create armor stand as intermediate carrier
        ArmorStandEntity stand = new ArmorStandEntity(EntityType.ARMOR_STAND, holder.getEntityWorld());
        stand.setInvisible(true);
        // stand.setMarker(true); // requires accesswidener
        stand.setNoGravity(true);
        stand.setInvulnerable(true);
        stand.setSilent(true);
        // stand.setSmall(true); // requires accesswidener

        Vec3d pos = holder.getEntityPos();
        stand.setPos(pos.x, pos.y, pos.z);
        (holder.getEntityWorld()).spawnEntity(stand);

        // Move held player to stand
        held.stopRiding();
        if (!held.startRiding(stand, true, false)) {
            // If the stand cannot accept the passenger, discard it and abort shield mode.
            stand.discard();
            return;
        }

        shieldStands.put(hid, stand);
        shieldModeActive.put(hid, true);
        shieldSwapCooldown.put(hid, TickClock.now() + SHIELD_SWAP_COOLDOWN_TICKS);

        Lang.sendActionBar(holder, "\u00a7aHuman shield activated!");
    }

    static void disableShieldMode(ServerPlayerEntity holder, boolean notify) {
        UUID hid = holder.getUuid();
        UUID tid = GrabYeetHandler.getHeldId(hid);
        if (tid == null) return;

        var server = holder.getEntityWorld().getServer();
        if (server == null) return;
        ArmorStandEntity stand = shieldStands.get(hid);

        if (stand != null) {
            if (stand.getFirstPassenger() != null) stand.getFirstPassenger().stopRiding();
            stand.discard();
            shieldStands.remove(hid);
        }

        // Never re-ride the holder: passenger sync would override the authoritative
        // requestTeleport applied by tickCarryPositions().
        shieldModeActive.put(hid, false);
        shieldSwapCooldown.put(hid, TickClock.now() + SHIELD_SWAP_COOLDOWN_TICKS);
        if (notify) {
            Lang.sendActionBar(holder, "§cHuman shield deactivated!");
        }
    }

    static boolean isActive(UUID holderId) {
        return Boolean.TRUE.equals(shieldModeActive.get(holderId));
    }

    }

    static void cleanup(UUID playerId) {

        if (stand != null) stand.discard();
        shieldModeActive.remove(playerId);
        shieldSwapCooldown.remove(playerId);
    }

    }

            if (stand != null) stand.discard();
        }
        shieldStands.clear();
        shieldModeActive.clear();
        shieldSwapCooldown.clear();
    }

        ServerLivingEntityEvents.ALLOW_DAMAGE.register((entity, source, amount) -> {

                shield.damage(holder.getEntityWorld(), source, amount);
                return false; // Cancel damage to holder, redirect to held player
            }
            return true;
        });

/** Shared cooldown plumbing; interaction-specific timing remains at the call site. */

    private InteractionCooldowns() {}

    public static boolean isBlocked(ServerPlayerEntity player, String type) {

                && CooldownManager.isOnCooldown(player.getUuid(), type);

    public static boolean rejectIfBlocked(ServerPlayerEntity player, String type) {
        if (!isBlocked(player, type)) return false;
        Lang.cooldown(player);

    }

    public static long apply(UUID playerId, String type, long configMs) {
        long ticks = AdminManager.resolveCooldownTicks(playerId, type, configMs);
        CooldownManager.setCooldown(playerId, type, ticks);
        return ticks;
    }

    /** Uses the initiator's override and applies the same duration to both participants. */
    public static void applyShared(UUID initiatorId, UUID targetId, String type, long configMs) {
        long ticks = apply(initiatorId, type, configMs);
        CooldownManager.setCooldown(targetId, type, ticks);
    }
}

package com.dap.dapurhomies.interaction;

import com.dap.dapurhomies.DapUrHomiesMod;
import com.dap.dapurhomies.compat.ProtectionCompat;
import com.dap.dapurhomies.config.ModConfig;
import com.dap.dapurhomies.permission.PlayerPermissionManager;
import com.dap.dapurhomies.util.TickClock;
import net.fabricmc.fabric.api.event.player.AttackEntityCallback;
import net.fabricmc.fabric.api.event.player.UseEntityCallback;
import net.minecraft.server.network.ServerPlayerEntity;
import net.minecraft.util.ActionResult;

import java.util.HashMap;
import java.util.Map;

public class InteractionRouter {
    // Debounce repeated right-click interaction chains per player pair and tick.
    private static final Map<Long, Long> lastRightClickTick = new HashMap<>();
    private static long lastCleanupTick = 0;
    public static void register() {
        // ===== RIGHT-CLICK INTERACTIONS =====
        UseEntityCallback.EVENT.register((player, world, hand, entity, hit) -> {
            if (world.isClient()) return ActionResult.PASS;
            if (!(player instanceof ServerPlayerEntity sp)) return ActionResult.PASS;
            if (!(entity instanceof ServerPlayerEntity target)) return ActionResult.PASS;
            if (!DapUrHomiesMod.isModEnabled()) return ActionResult.PASS;
            if (sp.getEntityWorld() != target.getEntityWorld()) return ActionResult.PASS;

            boolean isAdmin = PlayerPermissionManager.hasPermission(sp, "dap-ur-homies.admin");
            if (!PlayerPermissionManager.canInitiate(sp, "enabled")) return ActionResult.FAIL;
            if (!isAdmin && !PlayerPermissionManager.canReceive(target, "enabled")) return ActionResult.FAIL;
            if (!isAdmin && !ProtectionCompat.canInteract(sp, target)) return ActionResult.FAIL;

            // P0: holding -> yeet charge
            if (PlayerInteractionState.isHolding(sp) && ModConfig.get().enableGrabYeet)
                return GrabYeetHandler.startCharge(sp);

            if (PlayerInteractionState.isBusyForUse(sp, target))
                return ActionResult.PASS;

            boolean sneaking = sp.isSneaking();
            boolean sprinting = sp.isSprinting();
            boolean hitAtFeet = hit != null && hit.getPos().y < target.getY() + 0.8;
            double dist = sp.distanceTo(target);

            long nowTick = TickClock.now();
            // Cleanup expired debounce entries (older than 5 seconds = 100 ticks) — throttled to every 20 ticks
            if (nowTick - lastCleanupTick > 20) {
                lastRightClickTick.entrySet().removeIf(e -> nowTick - e.getValue() > 100);
                hugClickTick.entrySet().removeIf(e -> nowTick - e.getValue() > HUG_WINDOW_TICKS * 2);
                lastCleanupTick = nowTick;
            }
            long pairKey = (((long) sp.getUuid().hashCode()) << 32) | (target.getUuid().hashCode() & 0xffffffffL);
            Long lastRc = lastRightClickTick.get(pairKey);
            if (lastRc != null && nowTick - lastRc < 5) return ActionResult.PASS;

            // Hug: BOTH players sneak, aim at each other's HEAD and right-click within the window.
            // Must run BEFORE the sneak branch. If the partner does not answer, fall through to
            // charged Dap instead of swallowing the click entirely.
            if (sneaking && target.isSneaking() && ModConfig.get().enableHug
                    && dist <= ModConfig.get().hugRange
                    && aimingAtHead(sp, target) && aimingAtHead(target, sp)) {
                long key = hugKey(sp, target);
                Long last = hugClickTick.get(key);
                if (last != null && nowTick - last <= HUG_WINDOW_TICKS) {
                    ActionResult r = HugHandler.handleHug(sp, target);
                    if (r != ActionResult.PASS) {
                        // A pending charged Dap from the first click must not fire after the hug
                        DapHandler.cancelCharge(sp.getUuid());
                        DapHandler.cancelCharge(target.getUuid());
                        lastRightClickTick.put(pairKey, nowTick);
                        clearHugWindow(sp, target);
                        return r;
                    }
                }
                hugClickTick.put(key, nowTick); // first click: remember and keep processing
            }

            // SNEAK = charged dap (蓄力击掌) / charged DropKick when aiming at the feet
            if (sneaking) {
                if (hitAtFeet && ModConfig.get().enableKick) {
                    if (KickHandler.isCharging(sp.getUuid())) {
                        ActionResult r = KickHandler.releaseCharge(sp, target);
                        if (r != ActionResult.PASS) { lastRightClickTick.put(pairKey, nowTick); return r; }
                    } else {
                        KickHandler.startCharge(sp);
                        lastRightClickTick.put(pairKey, nowTick);
                        return ActionResult.SUCCESS;
                    }
                }
                if (ModConfig.get().enableDap && sp.getMainHandStack().isEmpty()) {
                    ActionResult r = DapHandler.startCharge(sp, target);
                    if (r != ActionResult.PASS) { lastRightClickTick.put(pairKey, nowTick); return r; }
                }
                return ActionResult.PASS;
            }

            // SPRINT = Push (疾跑推，不瞄脚部时)
            if (sprinting && !hitAtFeet) {
                if (ModConfig.get().enablePush) {
                    ActionResult r = PushHandler.handlePush(sp, target);
                    if (r != ActionResult.PASS) { lastRightClickTick.put(pairKey, nowTick); return r; }
                }
                return ActionResult.PASS;
            }

            // Kick (aim feet, not sneaking)
            if (hitAtFeet && ModConfig.get().enableKick) {
                ActionResult r = KickHandler.handleKick(sp, target);
                if (r != ActionResult.PASS) { lastRightClickTick.put(pairKey, nowTick); return r; }
            }
            // Slap (behind)
            if (ModConfig.get().enableSlap && isBehind(sp, target)) {
                ActionResult r = SimpleInteractions.handleSlap(sp, target);
                if (r != ActionResult.PASS) { lastRightClickTick.put(pairKey, nowTick); return r; }
            }
            // Right-click fallback = HighFive/Dap (击掌)
            if (sp.getMainHandStack().isEmpty() && ModConfig.get().enableDap) {
                ActionResult r = DapHandler.handleNormalDap(sp, target);
                if (r != ActionResult.PASS) { lastRightClickTick.put(pairKey, nowTick); return r; }
            }
            return ActionResult.PASS;
        });

        // ===== LEFT-CLICK INTERACTIONS (AttackEntityCallback) =====
        AttackEntityCallback.EVENT.register((player, world, hand, entity, hitResult) -> {
            if (world.isClient()) return ActionResult.PASS;
            if (!(player instanceof ServerPlayerEntity sp)) return ActionResult.PASS;
            if (!(entity instanceof ServerPlayerEntity target)) return ActionResult.PASS;
            if (!DapUrHomiesMod.isModEnabled()) return ActionResult.PASS;
            if (sp.getEntityWorld() != target.getEntityWorld()) return ActionResult.PASS;

            boolean isAdmin = PlayerPermissionManager.hasPermission(sp, "dap-ur-homies.admin");

            // Master switch check
            if (!PlayerPermissionManager.canInitiate(sp, "enabled")) return ActionResult.PASS;
            if (!isAdmin && !PlayerPermissionManager.canReceive(target, "enabled")) return ActionResult.PASS;
            if (!isAdmin && !ProtectionCompat.canInteract(sp, target)) return ActionResult.PASS;

            // State exclusion
            if (PlayerInteractionState.isBusyForAttack(sp, target))
                return ActionResult.PASS;

            // Shift+左键 = Grab (if not holding)
            if (sp.isSneaking() && ModConfig.get().enableGrabYeet
                    && !GrabYeetHandler.isHolding(sp) && !GrabYeetHandler.isHeld(sp)) {
                ActionResult result = GrabYeetHandler.tryGrab(sp, target);
                if (result != ActionResult.PASS) return result;
            }

            // left-click held target = throw if charged, else drop
            if (GrabYeetHandler.isHolding(sp) && ModConfig.get().enableGrabYeet) {
                if (GrabYeetHandler.isCharging(sp.getUuid())) return GrabYeetHandler.releaseCharge(sp);
                ActionResult r = GrabYeetHandler.tryDrop(sp);
                return (r == ActionResult.PASS) ? ActionResult.FAIL : r;
            }

            // Left-click does NOT trigger dap/highfive (only right-click does).
            return ActionResult.PASS;
        });
    }

    private static boolean isBehind(ServerPlayerEntity attacker, ServerPlayerEntity target) {
        net.minecraft.util.math.Vec3d look = target.getRotationVec(1.0f);
        net.minecraft.util.math.Vec3d to = attacker.getEntityPos().subtract(target.getEntityPos()).normalize();
        return look.dotProduct(to) <= 0.0;
    }

    // ---- Hug mutual-detection helpers ----
    private static final Map<Long, Long> hugClickTick = new HashMap<>(); // pairKey -> last tick initiator clicked
    private static final long HUG_WINDOW_TICKS = 10; // 0.5s: enough for two humans to answer each other

    private static long hugKey(ServerPlayerEntity a, ServerPlayerEntity b) {
        // order-independent pair key
        int h1 = a.getUuid().hashCode(), h2 = b.getUuid().hashCode();
        long lo = Math.min(h1, h2), hi = Math.max(h1, h2);
        return (hi << 32) | (lo & 0xffffffffL);
    }
    private static void clearHugWindow(ServerPlayerEntity a, ServerPlayerEntity b) {
        hugClickTick.remove(hugKey(a, b));
    }
    private static boolean aimingAtHead(ServerPlayerEntity viewer, ServerPlayerEntity target) {
        net.minecraft.util.math.Vec3d eye = viewer.getCameraPosVec(1.0f);
        net.minecraft.util.math.Vec3d look = viewer.getRotationVec(1.0f);
        net.minecraft.util.math.Vec3d head = target.getEntityPos().add(0, target.getStandingEyeHeight(), 0);
        net.minecraft.util.math.Vec3d toHead = head.subtract(eye).normalize();
        return look.dotProduct(toHead) > 0.95;   // ~18° cone on head
    }
}

    private static final Map<UUID, Long> chargeStartTime = new HashMap<>();

    private static final Map<UUID, Long> pushWindowEnd = new HashMap<>();

    private static final Identifier KICK_SLOWDOWN_ID = Identifier.of("dap-ur-homies", "kick_slowdown");
    private static final Identifier DROPKICK_SLOWDOWN_ID = Identifier.of("dap-ur-homies", "dropkick_slowdown");

    public static ActionResult handleKick(ServerPlayerEntity sp, ServerPlayerEntity target) {

        if (sp.equals(target)) return ActionResult.PASS;
        if (!ModConfig.get().enableKick) return ActionResult.PASS;
        if (sp.getEntityWorld() != target.getEntityWorld()) return ActionResult.PASS;

        double dist = sp.distanceTo(target);
        if (dist > ModConfig.get().kickRange) return ActionResult.PASS;

        // Look angle check: must be looking down at target's feet
        Vec3d look = sp.getRotationVec(1.0f);
        if (look.y > -0.25) return ActionResult.PASS;

        if (!PlayerPermissionManager.canInitiate(sp, "kick")) {
            Lang.interactionBlocked(sp, "Kick is disabled for you!", "你的踢功能已关闭！");
            return ActionResult.FAIL;
        }
        if (!PlayerPermissionManager.hasPermission(sp, "dap-ur-homies.admin")
                && !PlayerPermissionManager.canReceive(target, "kick")) {
            Lang.targetDisabledType(sp, target, "kick");
            return ActionResult.FAIL;
        }

        if (InteractionCooldowns.rejectIfBlocked(sp, "kick")) return ActionResult.FAIL;

        executeNormalKick(sp, target);
        return ActionResult.SUCCESS;

        if (player == null) return ActionResult.PASS;
        Long start = chargeStartTime.remove(player.getUuid());
        if (start == null) return ActionResult.PASS;
        if (!ModConfig.get().enableKick) return ActionResult.PASS;

        // The release must pass exactly the same gates as a normal kick / dropkick
        if (!PlayerPermissionManager.canInitiate(player, "kick")) {
            Lang.interactionBlocked(player, "Kick is disabled for you!", "你的踢功能已关闭！");
            return ActionResult.FAIL;

            return executeDropKick(player);

        if (target != null) {

    private static void executeNormalKick(ServerPlayerEntity sp, ServerPlayerEntity target) {
        if (sp == null || target == null) return;
        ServerWorld w = sp.getEntityWorld();
        if (w == null) return;

        UUID id = sp.getUuid();
        // Cooldown was checked by handleKick.

        Vec3d dir = target.getEntityPos().subtract(sp.getEntityPos()).normalize();
        if (dir.lengthSquared() < 0.001) dir = new Vec3d(1, 0, 0);

        // Apply speed slowdown
        applySlowdown(target, SLOWDOWN_AMOUNT, KICK_SLOWDOWN_ID);
        slowdownExpiry.put(target.getUuid(), TickClock.now() + SLOWDOWN_DURATION_TICKS);

        // Start push window
        pushWindowEnd.put(sp.getUuid(), TickClock.now() + TickClock.msToTicks(PUSH_WINDOW_MS));

        Vec3d mid = sp.getEntityPos().add(target.getEntityPos()).multiply(0.5).add(0, 0.5, 0);
        w.spawnParticles(ParticleTypes.SWEEP_ATTACK, mid.x, mid.y, mid.z, 3, 0.2, 0.1, 0.2, 0.05);

        com.dap.dapurhomies.compat.DanseCompat.playKick(sp, target); // Danse animation
    }

    // ===== DROP KICK =====

    private static ActionResult executeDropKick(ServerPlayerEntity sp) {
        if (sp == null) return ActionResult.PASS;
        ServerWorld w = sp.getEntityWorld();
        if (w == null) return ActionResult.PASS;

        UUID id = sp.getUuid();
        // Cooldown was checked by releaseCharge.

        // DropKick affects all eligible living entities in front of the player.
        Vec3d eyePos = sp.getEyePos();
        Vec3d look = sp.getRotationVec(1.0f);

        boolean hit = false;
        Box box = sp.getBoundingBox().expand(DROPKICK_RANGE + 0.5);
        for (Entity entity : sp.getEntityWorld().getOtherEntities(sp, box,
                e -> e instanceof LivingEntity && !e.isRemoved() && !e.isSpectator())) {
            LivingEntity target = (LivingEntity) entity;
            if (target == sp) continue;
            if (target.getEntityWorld() != sp.getEntityWorld()) continue;
            if (target.getEntityPos().squaredDistanceTo(eyePos) > DROPKICK_RANGE * DROPKICK_RANGE) continue;

            // Player targets respect their receive toggle and protection claims
            if (target instanceof ServerPlayerEntity playerTarget
                    && !PlayerPermissionManager.hasPermission(sp, "dap-ur-homies.admin")) {
                if (!PlayerPermissionManager.canReceive(playerTarget, "kick")) continue;
                if (!com.dap.dapurhomies.compat.ProtectionCompat.canInteract(sp, playerTarget)) continue;
            }

            Vec3d toTarget = target.getEntityPos().subtract(eyePos).normalize();
            double dot = look.dotProduct(toTarget);
            if (dot < 0.5) continue; // Must be roughly in front

            Vec3d dir = target.getEntityPos().subtract(sp.getEntityPos()).normalize();
            if (dir.lengthSquared() < 0.001) dir = new Vec3d(1, 0, 0);

            target.addVelocity(dir.x * DROPKICK_KNOCKBACK, 0.5, dir.z * DROPKICK_KNOCKBACK); // ~5 blocks high
            target.knockedBack = true;
            target.damage(w, w.getDamageSources().create(DamageTypes.GENERIC, sp), (float) DROPKICK_DAMAGE);

            // Apply heavy slowdown (only for players)
            if (target instanceof ServerPlayerEntity playerTarget) {
                applySlowdown(playerTarget, DROPKICK_SLOWDOWN, DROPKICK_SLOWDOWN_ID);
                slowdownExpiry.put(playerTarget.getUuid(), TickClock.now() + SLOWDOWN_DURATION_TICKS);
            }

            // Start push window
            pushWindowEnd.put(sp.getUuid(), TickClock.now() + TickClock.msToTicks(PUSH_WINDOW_MS));

            Vec3d mid = sp.getEntityPos().add(target.getEntityPos()).multiply(0.5).add(0, 0.5, 0);
            w.spawnParticles(ParticleTypes.EXPLOSION, mid.x, mid.y, mid.z, 5, 0.3, 0.3, 0.3, 0.1);
            w.playSound(null, mid.x, mid.y, mid.z, SoundEvents.ENTITY_GENERIC_EXPLODE.value(), SoundCategory.PLAYERS, 1.5f, 0.8f);

            hit = true;
        }

        if (hit) {
            sp.swingHand(net.minecraft.util.Hand.MAIN_HAND, true);
            Lang.sendActionBar(sp, "§c DROP KICK!", "§c 飞踢!");
            InteractionCooldowns.apply(sp.getUuid(), "kick", ModConfig.get().kickCooldownMs * 2);
            return ActionResult.SUCCESS;
        }
        return ActionResult.PASS;
    }

    // ===== PUSH WINDOW TICK =====

    private static void tickPushWindow(MinecraftServer server) {
        long now = TickClock.now();
        Iterator<Map.Entry<UUID, Long>> it = pushWindowEnd.entrySet().iterator();
        while (it.hasNext()) {
            Map.Entry<UUID, Long> entry = it.next();
            if (now >= entry.getValue()) {
                it.remove();
                continue;
            }

            ServerPlayerEntity kicker = server.getPlayerManager().getPlayer(entry.getKey());
            if (kicker == null || !kicker.isAlive()) {
                it.remove();
                continue;
            }

            // Apply small continuous push to entities in front
            Vec3d eyePos = kicker.getEyePos();
            Vec3d look = kicker.getRotationVec(1.0f);
            for (ServerPlayerEntity target : server.getPlayerManager().getPlayerList()) {
                if (target == kicker) continue;
                if (target.getEntityWorld() != kicker.getEntityWorld()) continue;
                if (target.getEntityPos().squaredDistanceTo(eyePos) > 4.0) continue;

                Vec3d toTarget = target.getEntityPos().subtract(eyePos).normalize();
                if (look.dotProduct(toTarget) < 0.7) continue;

                Vec3d pushDir = target.getEntityPos().subtract(kicker.getEntityPos()).normalize();
                target.addVelocity(pushDir.x * 0.15, 0.05, pushDir.z * 0.15);
                target.knockedBack = true;
            }
        }
    }

    // ===== SLOWDOWN =====

    private static void applySlowdown(ServerPlayerEntity player, double amount, Identifier id) {
        EntityAttributeInstance speedAttr = player.getAttributeInstance(EntityAttributes.MOVEMENT_SPEED);
        if (speedAttr != null) {
            // Remove existing modifier
            speedAttr.removeModifier(id);
            // Add new modifier
            EntityAttributeModifier modifier = new EntityAttributeModifier(
                    id, amount, EntityAttributeModifier.Operation.ADD_VALUE
            );
            speedAttr.addTemporaryModifier(modifier);
        }
    }

    public static void removeSlowdown(ServerPlayerEntity player, Identifier id) {
        EntityAttributeInstance speedAttr = player.getAttributeInstance(EntityAttributes.MOVEMENT_SPEED);
        if (speedAttr != null) speedAttr.removeModifier(id);
    }

    // ===== TICK =====

    /** Auto-release charges once full so a stale charge can never fire later. */
    private static void tickCharges(MinecraftServer server) {
        long now = TickClock.now();
        long fullTicks = TickClock.msToTicks(CHARGE_DURATION_MS);
        List<UUID> ready = new ArrayList<>();
        for (Map.Entry<UUID, Long> e : chargeStartTime.entrySet()) {
            if (now - e.getValue() >= fullTicks) ready.add(e.getKey());
        }
        for (UUID id : ready) {
            ServerPlayerEntity player = server.getPlayerManager().getPlayer(id);
            if (player == null) {
                cancelCharge(id);
                continue;
            }
            releaseCharge(player);
        }
    }

    public static void tick(MinecraftServer server) {
        tickPushWindow(server);
        tickCharges(server);
        long now = TickClock.now();
        Iterator<Map.Entry<UUID, Long>> slowIt = slowdownExpiry.entrySet().iterator();
        while (slowIt.hasNext()) {
            Map.Entry<UUID, Long> e = slowIt.next();
            if (now >= e.getValue()) {
                ServerPlayerEntity p = server.getPlayerManager().getPlayer(e.getKey());
                if (p != null) {
                    removeSlowdown(p, KICK_SLOWDOWN_ID);
                    removeSlowdown(p, DROPKICK_SLOWDOWN_ID);
                }
                slowIt.remove();
            }
        }
    }

    // ===== CLEANUP =====

    public static void cleanup(UUID playerId) {
        chargeStartTime.remove(playerId);
        pushWindowEnd.remove(playerId);
        slowdownExpiry.remove(playerId);
    }

        // Handled by CooldownManager
    }
}

import net.minecraft.particle.ParticleTypes;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.network.ServerPlayerEntity;

import java.util.HashMap;

import java.util.Map;

/**
 * Tracks launched/thrown players and plays trail particle effects.
 * Ported from original Dap-ur-homie project.
 */
public class LaunchedPlayerTracker {

    private static final Map<UUID, Integer> launchedTicks = new HashMap<>();
    private static final int TRAIL_DURATION = 20; // 1 second of particle trail

    public static void markPlayerAsLaunched(UUID playerId) {
        launchedTicks.put(playerId, 0);
    }

    public static void tick(MinecraftServer server) {
        Iterator<Map.Entry<UUID, Integer>> it = launchedTicks.entrySet().iterator();
        while (it.hasNext()) {
            Map.Entry<UUID, Integer> entry = it.next();
            UUID id = entry.getKey();
            int ticks = entry.getValue();

            ServerPlayerEntity player = server.getPlayerManager().getPlayer(id);
            if (player == null || !player.isAlive()) {
                it.remove();
                continue;
            }

            if (ticks < TRAIL_DURATION) {
                playLaunchTrailEffects(player);
                entry.setValue(ticks + 1);
            } else {
                it.remove();
            }
        }
    }

    private static void playLaunchTrailEffects(ServerPlayerEntity player) {
        var world = player.getEntityWorld();
        var pos = player.getEntityPos();

        // Cloud trail
        world.spawnParticles(ParticleTypes.CLOUD,
                pos.x, pos.y + 0.5, pos.z,
                2, 0.15, 0.15, 0.15, 0.02);

        // Crit sparkles for high-speed flight
        if (player.getVelocity().lengthSquared() > 1.0) {
            world.spawnParticles(ParticleTypes.CRIT,
                    pos.x, pos.y + 0.5, pos.z,
                    1, 0.1, 0.1, 0.1, 0.05);
        }
    }

    public static void cleanup(UUID playerId) {
        launchedTicks.remove(playerId);
    }
}

/**
 * Facade for cross-module player interaction state queries.
 * Callers (router, main tick loop) depend on this single entry point
 * instead of reaching into every handler's static state maps.
 */

    private PlayerInteractionState() {}

        return GrabYeetHandler.hasAnyHolding();
    }

    }

    public static boolean isHugging(UUID id) {
        return HugHandler.isHugging(id);
    }

    public static boolean isInDapSession(UUID id) {

    }

    /** Right-click exclusion set (mirrors the router's historical semantics). */
    public static boolean isBusyForUse(ServerPlayerEntity self, ServerPlayerEntity target) {

/**
 * Push system - ported from original Dap-ur-homie project.
 * Features: push immunity, ceiling detection, tiered push power.
 */
public class PushHandler {

    private static final Map<UUID, Long> pushImmunity = new HashMap<>();
    private static final long PUSH_IMMUNITY_TICKS = TickClock.msToTicks(500);

    // Unified push velocity: ~5 blocks high, 8 blocks far
    private static final double PUSH_VELOCITY = 1.2;

    public static ActionResult handlePush(ServerPlayerEntity sp, ServerPlayerEntity target) {
        if (sp == null || target == null) return ActionResult.PASS;
        if (sp.equals(target)) return ActionResult.PASS;
        if (!ModConfig.get().enablePush) return ActionResult.PASS;

        double dist = sp.distanceTo(target);
        if (dist > ModConfig.get().pushRange) return ActionResult.PASS;

        if (!PlayerPermissionManager.canInitiate(sp, "push")) {
            Lang.interactionBlocked(sp, "Push is disabled for you!", "你的推功能已关闭！");
            return ActionResult.FAIL;
        }
        if (!PlayerPermissionManager.hasPermission(sp, "dap-ur-homies.admin")
                && !PlayerPermissionManager.canReceive(target, "push")) {
            Lang.targetDisabledType(sp, target, "push");
            return ActionResult.FAIL;
        }

        if (InteractionCooldowns.rejectIfBlocked(sp, "push")) return ActionResult.FAIL;

        if (isImmune(target.getUuid())) return ActionResult.PASS;

        executePush(sp, target);
        return ActionResult.SUCCESS;
    }

    private static void executePush(ServerPlayerEntity sp, ServerPlayerEntity target) {
        if (sp == null || target == null) return;
        applyPush(sp, target, PUSH_VELOCITY);
    }

    private static void applyPush(ServerPlayerEntity sp, ServerPlayerEntity target, double velocity) {
        if (sp == null || target == null) return;
        ServerWorld w = sp.getEntityWorld();
        if (w == null) return;

        // Original project: target.addVelocity(0, velocity, 0) - vertical launch
        double ceilingHeight = getCeilingHeight(target);
        double actualVelocity = velocity;
        if (ceilingHeight < 10 && ceilingHeight > 0) {
            double maxH = Math.max(2, ceilingHeight - 1);
            double v = Math.sqrt(2 * 0.08 * 20 * maxH);
            actualVelocity = Math.min(v, velocity);
        }

        UUID carriedId = GrabYeetHandler.getHeldId(target.getUuid());
        if (carriedId != null) {
            var srv = w.getServer();
            if (srv != null) {
                ServerPlayerEntity carried = srv.getPlayerManager().getPlayer(carriedId);
                if (carried != null) {
                    carried.addVelocity(0, actualVelocity + 0.2, 0);
                    carried.knockedBack = true;
                    pushImmunity.put(carriedId, TickClock.now() + PUSH_IMMUNITY_TICKS);
                }
            }
        }

        // Calculate horizontal away direction for knockback
        Vec3d awayDir = target.getEntityPos().subtract(sp.getEntityPos()).normalize();
        awayDir = new Vec3d(awayDir.x, 0, awayDir.z).normalize();
        if (awayDir.lengthSquared() < 0.001) awayDir = new Vec3d(1, 0, 0);

        target.addVelocity(awayDir.x * actualVelocity * 0.8, actualVelocity * 0.6, awayDir.z * actualVelocity * 0.8);
        target.knockedBack = true;

        pushImmunity.put(target.getUuid(), TickClock.now() + PUSH_IMMUNITY_TICKS);

        Vec3d mid = sp.getEntityPos().add(target.getEntityPos()).multiply(0.5).add(0, 1.2, 0);
        w.spawnParticles(ParticleTypes.CLOUD, mid.x, mid.y, mid.z, 6, 0.3, 0.2, 0.3, 0.05);
        w.playSound(null, mid.x, mid.y, mid.z, SoundEvents.ENTITY_PLAYER_ATTACK_KNOCKBACK, SoundCategory.PLAYERS, 1.0f, 1.0f);

        sp.swingHand(net.minecraft.util.Hand.MAIN_HAND, true);
        Lang.pushSuccess(sp, target);
        InteractionCooldowns.applyShared(sp.getUuid(), target.getUuid(), "push", ModConfig.get().pushCooldownMs);
        com.dap.dapurhomies.compat.DanseCompat.playPush(sp, target); // Danse animation
    }

    private static double getCeilingHeight(ServerPlayerEntity player) {
        ServerWorld world = player.getEntityWorld();
        BlockPos pos = BlockPos.ofFloored(player.getEntityPos());
        int maxCheck = 20;
        for (int i = 1; i <= maxCheck; i++) {
            if (!world.getBlockState(pos.up(i)).isAir()) return i;
        }
        return maxCheck;
    }

    private static boolean isImmune(UUID playerId) {
        Long endTime = pushImmunity.get(playerId);
        return endTime != null && TickClock.now() < endTime;
    }

    public static void tick(MinecraftServer server) {
        cleanupExpired();
    }

    private static void cleanupExpired() {
        long now = TickClock.now();
        pushImmunity.entrySet().removeIf(e -> now >= e.getValue());
    }

    public static void cleanup(UUID playerId) {
        pushImmunity.remove(playerId);
    }

    public static void purgeExpiredCooldowns() {
        cleanupExpired();
    }
}

package com.dap.dapurhomies.interaction;

import com.dap.dapurhomies.compat.ProtectionCompat;
import com.dap.dapurhomies.util.TickClock;
import net.minecraft.block.BlockState;
import net.minecraft.block.Blocks;
import net.minecraft.particle.ParticleTypes;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.network.ServerPlayerEntity;
import net.minecraft.server.world.ServerWorld;
import net.minecraft.sound.SoundCategory;
import net.minecraft.sound.SoundEvents;
import net.minecraft.util.math.BlockPos;
import net.minecraft.util.math.Vec3d;
import net.minecraft.world.World;

import java.util.HashMap;
import java.util.Iterator;
import java.util.Map;
import java.util.UUID;

final class ThrowTracker {
    private static final Map<UUID, PendingThrow> pendingThrows = new HashMap<>();

    private static class PendingThrow {
        final Vec3d velocity;
        int ticksRemaining;

        PendingThrow(Vec3d velocity, int delayTicks) {
            this.velocity = velocity;
            this.ticksRemaining = delayTicks;
        }
    }

    private static final Map<UUID, ThrownPlayerData> thrownPlayers = new HashMap<>();

    private static class ThrownPlayerData {
        final long throwTick;
        boolean wasOnFire;
        boolean hitWall;
        boolean landed;
        int ticksFlying;

        ThrownPlayerData(long throwTick, boolean wasOnFire) {
            this.throwTick = throwTick;
            this.wasOnFire = wasOnFire;
            this.hitWall = false;
            this.landed = false;
            this.ticksFlying = 0;
        }
    }

    private static final Map<UUID, Long> throwImmunity = new HashMap<>();
    private static final long THROW_IMMUNITY_TICKS = 10;      // 500ms = 10 ticks

    private static final int PENDING_THROW_TICKS = 3;
    private static final double WALL_CHECK_DISTANCE = 1.5;
    private static final float FIRE_EXPLOSION_POWER = 3.0f;

    private ThrowTracker() {
    }

    static boolean isImmune(UUID playerId) {
        Long immuneTick = throwImmunity.get(playerId);
        return immuneTick != null && (TickClock.now() - immuneTick) < 20;
    }

    static void schedulePendingThrow(UUID playerId, Vec3d velocity) {
        pendingThrows.put(playerId, new PendingThrow(velocity, PENDING_THROW_TICKS));
    }

    static void registerThrown(UUID playerId, boolean wasOnFire) {
        thrownPlayers.put(playerId, new ThrownPlayerData(TickClock.now(), wasOnFire));
    }

    static void applyImmunity(UUID playerId) {
        throwImmunity.put(playerId, TickClock.now());
    }

    static boolean isThrownFlying(UUID playerId) {
        return thrownPlayers.containsKey(playerId);
    }

    static void tickPendingThrows(MinecraftServer server) {
        Iterator<Map.Entry<UUID, PendingThrow>> it = pendingThrows.entrySet().iterator();
        while (it.hasNext()) {
            Map.Entry<UUID, PendingThrow> entry = it.next();
            UUID playerId = entry.getKey();
            PendingThrow pt = entry.getValue();
            pt.ticksRemaining--;

            if (pt.ticksRemaining <= 0) {
                ServerPlayerEntity player = server.getPlayerManager().getPlayer(playerId);
                if (player != null && player.isAlive()) {
                    player.setVelocity(pt.velocity);
                    player.knockedBack = true;
                }
                it.remove();
            }
        }
    }

    static void tickThrownPlayers(MinecraftServer server) {
        long nowTick = TickClock.now();
        Iterator<Map.Entry<UUID, ThrownPlayerData>> it = thrownPlayers.entrySet().iterator();

        while (it.hasNext()) {
            Map.Entry<UUID, ThrownPlayerData> entry = it.next();
            UUID playerId = entry.getKey();
            ThrownPlayerData data = entry.getValue();
            ServerPlayerEntity player = server.getPlayerManager().getPlayer(playerId);

            if (player == null || !player.isAlive()) {
                it.remove();
                continue;
            }

            Vec3d pos = player.getEntityPos();
            ServerWorld world = player.getEntityWorld();
            data.ticksFlying++;

            // Wall collision check (every 2 ticks to reduce block state queries)
            if (!data.hitWall && data.ticksFlying % 2 == 0) {
                Vec3d vel = player.getVelocity();
                Vec3d checkDir = (vel.lengthSquared() > 0.001) ? vel.normalize() : player.getRotationVec(1.0f);
                Vec3d checkPos = pos.add(checkDir.multiply(WALL_CHECK_DISTANCE));
                BlockPos blockPos = BlockPos.ofFloored(checkPos);
                BlockState state = world.getBlockState(blockPos);

                if (!state.isAir() && isBreakable(state)) {
                    // Ignore the first 3 ticks, while velocity direction is still unreliable.
                    // Break walls only at flight speeds of at least 1.2 blocks/tick.
                    if (data.ticksFlying > 3 && player.getVelocity().length() >= 1.2) {
                        data.hitWall = true;
                        // Respect claims: the thrown player must not be able to breach protected blocks
                        if (ProtectionCompat.canBuildAt(player, blockPos)) {
                            world.breakBlock(blockPos, true);
                        }
                        float hardness = state.getHardness(world, blockPos);
                        double speed = player.getVelocity().length();
                        float damage = calculateWallDamage(hardness, speed);
                        player.damage(world, world.getDamageSources().flyIntoWall(), damage);
                        player.setVelocity(player.getVelocity().multiply(0.7));
                        player.knockedBack = true;
                        world.spawnParticles(ParticleTypes.EXPLOSION, checkPos.x, checkPos.y, checkPos.z,
                                3, 0.3, 0.3, 0.3, 0.1);
                        world.playSound(null, checkPos.x, checkPos.y, checkPos.z,
                                SoundEvents.BLOCK_STONE_BREAK, SoundCategory.BLOCKS, 1.0f, 0.8f);
                    }
                }
            }

            // Landing detection
            if (!data.landed) {
                boolean onGround = player.isOnGround();
                boolean inWater = player.isTouchingWater();
                boolean nearGround = !world.getBlockState(BlockPos.ofFloored(pos.x, pos.y - 0.5, pos.z)).isAir();

                if (onGround || inWater || nearGround) {
                    data.landed = true;

                    // Fire explosion if was on fire
                    if (data.wasOnFire) {
                        world.createExplosion(null, pos.x, pos.y, pos.z,
                                FIRE_EXPLOSION_POWER, World.ExplosionSourceType.NONE);
                        world.spawnParticles(ParticleTypes.LAVA, pos.x, pos.y, pos.z,
                                20, 0.5, 0.5, 0.5, 0.1);
                        world.playSound(null, pos.x, pos.y, pos.z,
                                SoundEvents.ENTITY_GENERIC_EXPLODE.value(), SoundCategory.BLOCKS, 1.5f, 1.0f);
                    }

                    // Landing particles
                    world.spawnParticles(ParticleTypes.CLOUD, pos.x, pos.y, pos.z,
                            5, 0.3, 0.1, 0.3, 0.05);

                    it.remove();
                    continue;
                }
            }

            // Expire throws after 10 seconds (200 ticks).
            if (nowTick - data.throwTick > 200) {
                it.remove();
            }
        }
    }

    static void cleanupExpiredImmunity() {
        long nowTick = TickClock.now();
        throwImmunity.entrySet().removeIf(e -> nowTick - e.getValue() > THROW_IMMUNITY_TICKS);
    }

    private static float calculateWallDamage(float hardness, double speed) {
        float baseDamage = hardness * 0.8f;
        float speedMultiplier = (float) Math.min(2.0, speed);
        return Math.max(1.0f, Math.min(10.0f, baseDamage * speedMultiplier));
    }

    private static boolean isBreakable(BlockState state) {
        // Exclude unbreakable blocks
        return !state.isOf(Blocks.BEDROCK)
                && !state.isOf(Blocks.OBSIDIAN)
                && !state.isOf(Blocks.CRYING_OBSIDIAN)
                && !state.isOf(Blocks.REINFORCED_DEEPSLATE)
                && !state.isOf(Blocks.CHEST)
                && !state.isOf(Blocks.TRAPPED_CHEST)
                && !state.isOf(Blocks.ENDER_CHEST)
                && !state.isOf(Blocks.BARREL)
                && !state.isOf(Blocks.SHULKER_BOX)
                && !state.isOf(Blocks.BEACON)
                && !state.isOf(Blocks.COMMAND_BLOCK)
                && !state.isOf(Blocks.CHAIN_COMMAND_BLOCK)
                && !state.isOf(Blocks.REPEATING_COMMAND_BLOCK);
    }

    static void cleanup(UUID playerId) {
        pendingThrows.remove(playerId);
        thrownPlayers.remove(playerId);
        throwImmunity.remove(playerId);
    }

    static void fullCleanup() {
        pendingThrows.clear();
        thrownPlayers.clear();
        throwImmunity.clear();
    }

    static void purgeExpiredCooldowns() {
        cleanupExpiredImmunity();
    }
}

public abstract class GrabbedPlayerControlMixin {

    @Inject(method = "travel", at = @At("HEAD"), cancellable = true)
    private void dapurhomies$ignoreMovement(Vec3d movementInput, CallbackInfo ci) {

        if (GrabYeetHandler.isHeld(self.getUuid())) {
            ci.cancel();   // discard movement; rotation (yaw/pitch packets) still apply
        }
    }
}

package com.dap.dapurhomies;

import com.dap.dapurhomies.interaction.DapSessionManager;
import com.dap.dapurhomies.interaction.GrabYeetHandler;
import com.dap.dapurhomies.interaction.InteractionModules;
import com.dap.dapurhomies.interaction.PlayerInteractionState;
import com.dap.dapurhomies.util.TickClock;
import net.fabricmc.fabric.api.event.lifecycle.v1.ServerTickEvents;
import net.fabricmc.fabric.api.networking.v1.ServerPlayConnectionEvents;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.network.ServerPlayerEntity;

import java.util.Set;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

public class ModLifecycle {
    private static int purgeTickCounter = 0;
    private static final Set<UUID> handSwingingLastTick = ConcurrentHashMap.newKeySet();
    private static final Set<UUID> jumpHeldLastTick = ConcurrentHashMap.newKeySet();

    static void register() {
        ServerPlayConnectionEvents.DISCONNECT.register((handler, server) -> {
            UUID id = handler.player.getUuid();

            UUID heldId = GrabYeetHandler.getHeldId(id);
            if (heldId != null) {
                ServerPlayerEntity held = server.getPlayerManager().getPlayer(heldId);
                if (held != null) {
                    held.stopRiding();
                    Lang.sendActionBar(held, "Your carrier disconnected!", "携带你的玩家已断开连接！");
                }
            }

            // If disconnecting player is being held, notify their holder
            UUID holderId = GrabYeetHandler.getHolderId(id);
            if (holderId != null) {
                ServerPlayerEntity holder = server.getPlayerManager().getPlayer(holderId);
                if (holder != null) {
                    Lang.sendActionBar(holder, "Your held player disconnected!", "你携带的玩家已断开连接！");
                }
            }

            InteractionModules.cleanupAll(id);
            handSwingingLastTick.remove(id);
            jumpHeldLastTick.remove(id);
        });

        ServerTickEvents.END_SERVER_TICK.register(server -> {
            DapUrHomiesMod.serverInstance = server;
            TickClock.update(server);
            if (!DapUrHomiesMod.modEnabled) return;
            InteractionModules.tickAll(server);
            boolean hasAnyHolding = PlayerInteractionState.hasAnyHolding();
            for (var player : server.getPlayerManager().getPlayerList()) {
                // Detect a fresh jump while being held -> attempt escape (documented feature)
                if (GrabYeetHandler.isHeld(player)) {
                    boolean jumping = player.getPlayerInput().jump();
                    boolean wasJumping = jumpHeldLastTick.contains(player.getUuid());
                    if (jumping && !wasJumping) {
                        GrabYeetHandler.tryEscape(player);
                    }
                    if (jumping) jumpHeldLastTick.add(player.getUuid());
                    else jumpHeldLastTick.remove(player.getUuid());
                } else {
                    jumpHeldLastTick.remove(player.getUuid());
                }

                // Detect left-click while holding someone (handSwinging rising edge)
                // This catches left-clicks on AIR (AttackEntityCallback only fires on entities)
                // Skip check if nobody is holding anyone (performance optimization)
                if (hasAnyHolding && GrabYeetHandler.isHolding(player)) {
                    boolean isSwinging = player.handSwinging;
                    boolean wasSwinging = handSwingingLastTick.contains(player.getUuid());
                    if (isSwinging && !wasSwinging && !GrabYeetHandler.justGrabbed(player.getUuid())) {
                        if (GrabYeetHandler.isCharging(player.getUuid())) {
                            GrabYeetHandler.releaseCharge(player);   // charged -> yeet
                        } else {
                            GrabYeetHandler.tryDrop(player);          // not charged -> drop
                        }
                    }
                    if (isSwinging) handSwingingLastTick.add(player.getUuid());
                    else handSwingingLastTick.remove(player.getUuid());
                } else {
                    handSwingingLastTick.remove(player.getUuid());
                }

                // Mahito effect: random erratic soul-distortion movement
                if (player.hasStatusEffect(DapUrHomiesMod.MAHITO_EFFECT_ENTRY)) {
                    var inst = player.getStatusEffect(DapUrHomiesMod.MAHITO_EFFECT_ENTRY);
                    int amp = (inst != null) ? inst.getAmplifier() : 0;
                    if (player.getRandom().nextFloat() < 0.08f) {
                        double s = 0.15 + amp * 0.05;
                        player.addVelocity(
                            (player.getRandom().nextDouble() - 0.5) * s,
                            0.08 + player.getRandom().nextDouble() * 0.05,
                            (player.getRandom().nextDouble() - 0.5) * s
                        );
                        player.knockedBack = true; // Sync velocity to the client.
                    }
                }
            }
            if (++purgeTickCounter >= 200) {
                purgeTickCounter = 0;
                InteractionModules.purgeAll();
            }
        });
    }

    /**
     * Turning the mod off must also release every piece of live state. Otherwise a player
     * who is being carried stays frozen forever: tickAll() is skipped, but the client-side
     * travel cancellation mixin keeps suppressing their movement.
     */
    public static void disableModAndCleanup(ServerPlayerEntity actor) {
        DapUrHomiesMod.modEnabled = false;
        MinecraftServer server = actor != null ? actor.getEntityWorld().getServer() : DapUrHomiesMod.serverInstance;
        if (server != null) {
            for (ServerPlayerEntity online : server.getPlayerManager().getPlayerList()) {
                UUID id = online.getUuid();
                InteractionModules.cleanupAll(id);
                handSwingingLastTick.remove(id);
                jumpHeldLastTick.remove(id);
            }
        }
        GrabYeetHandler.fullCleanup();
        DapSessionManager.clearAll();
        if (actor != null) Lang.modDisabled(actor);
    }

public class FilePermissionProvider implements PermissionProvider {
    private static final Gson GSON = new GsonBuilder().setPrettyPrinting().create();
    private static final File FILE = FabricLoader.getInstance().getConfigDir().resolve("dap-ur-homies-perms.json").toFile();
    private final Map<UUID, PlayerPreferences> perms = new ConcurrentHashMap<>();

    public FilePermissionProvider() {
        load();
    }

    @Override
    public boolean hasPermission(ServerPlayerEntity player, String node) {
        if (node.equals("dap-ur-homies.admin")) {
            var perms = player.getCommandSource().getPermissions();
            if (perms instanceof net.minecraft.command.permission.LeveledPermissionPredicate) {
                return ((net.minecraft.command.permission.LeveledPermissionPredicate) perms).getLevel().getLevel() >= 2;
            }
            return false;
        }
        return true;
    }

    @Override
    public boolean canInitiate(ServerPlayerEntity player, String type) {
        return get(player.getUuid()).allows(type);
    }

    @Override
    public boolean canReceive(ServerPlayerEntity player, String type) {
        return get(player.getUuid()).allows(type);
    }

    @Override
    public void setPreference(ServerPlayerEntity player, String type, boolean value) {
        get(player.getUuid()).set(type, value);
        save();
    }

    @Override
    public PlayerPreferences getPerms(ServerPlayerEntity player) {
        return get(player.getUuid());
    }

    @Override
    public PlayerPreferences getPerms(UUID playerId) {
        return get(playerId);
    }

    @Override
    public String getLang(UUID playerId) {
        PlayerPreferences p = get(playerId);
        return p.lang != null ? p.lang : "global";
    }

    @Override
    public void setLang(UUID playerId, String lang) {
        PlayerPreferences p = get(playerId);
        p.lang = lang;
        save();
    }

    private PlayerPreferences get(UUID playerId) {
        return perms.computeIfAbsent(playerId, k -> new PlayerPreferences());
    }

    public void load() {
        if (!FILE.exists()) return;
        try (FileReader r = new FileReader(FILE)) {
            Type type = new TypeToken<Map<String, PlayerPreferences>>(){}.getType();
            Map<String, PlayerPreferences> raw = GSON.fromJson(r, type);
            if (raw != null) {
                perms.clear();
                for (Map.Entry<String, PlayerPreferences> e : raw.entrySet()) {
                    if (e.getValue() == null) continue;
                    try {
                        perms.put(UUID.fromString(e.getKey()), e.getValue());
                    } catch (IllegalArgumentException ignored) {
                        System.err.println("[DapUrHomies] Skipping invalid UUID in permissions: " + e.getKey());
                    }
                }
            }
        } catch (IOException | RuntimeException e) {
            System.err.println("[DapUrHomies] Failed to load permissions: " + e.getMessage());
        }
    }

    public void save() {
        try {
            FILE.getParentFile().mkdirs();
            Map<String, PlayerPreferences> raw = new ConcurrentHashMap<>();
            for (Map.Entry<UUID, PlayerPreferences> e : perms.entrySet()) {
                raw.put(e.getKey().toString(), e.getValue());
            }
            try (FileWriter w = new FileWriter(FILE)) {
                GSON.toJson(raw, w);
            }
        } catch (IOException e) {
            System.err.println("[DapUrHomies] Failed to save permissions: " + e.getMessage());
        }
    }
}

package com.dap.dapurhomies.permission;

import com.dap.dapurhomies.config.ModConfig;
import net.minecraft.server.network.ServerPlayerEntity;

import java.lang.reflect.Method;

/**
 * LuckPerms / Fabric Permissions API bridge.
 * Uses reflection so LuckPerms is an optional dependency.
 *
 * Behaviour:
 *   - canInitiate:  LuckPerms node check  +  file preference fallback
 *   - canReceive:   file preference only by default;
 *                   if ModConfig.luckPermsControlsReceive = true,
 *                   also checks LuckPerms node "dap-ur-homies.receive.<type>"
 */
public class LuckPermsPermissionProvider implements PermissionProvider {
    private final FilePermissionProvider fallback = new FilePermissionProvider();
    private Method checkMethod;
    private boolean checkByEntity;
    private boolean available = false;

    public LuckPermsPermissionProvider() {
        try {
            Class<?> permissionsClass = Class.forName("me.lucko.fabric.api.permissions.v0.Permissions");
            try {
                // 当前 Fabric Permissions API 直接接受 Entity，避免命令源接口在不同映射下名称不同。
                checkMethod = permissionsClass.getMethod("check", net.minecraft.entity.Entity.class, String.class, boolean.class);
                checkByEntity = true;
            } catch (NoSuchMethodException ignored) {
                // 兼容旧版 API。
                checkMethod = permissionsClass.getMethod("check", net.minecraft.server.command.ServerCommandSource.class, String.class, boolean.class);
                checkByEntity = false;
            }
            available = true;
        } catch (Exception e) {
            available = false;
        }
    }

    public boolean isAvailable() {
        return available;
    }

    @Override
    public boolean hasPermission(ServerPlayerEntity player, String node) {
        if (!available) return fallback.hasPermission(player, node);
        try {
            Object source = checkByEntity ? player : player.getCommandSource();
            return (boolean) checkMethod.invoke(null, source, node, fallback.hasPermission(player, node));
        } catch (Exception e) {
            return fallback.hasPermission(player, node);
        }
    }

    @Override
    public boolean canInitiate(ServerPlayerEntity player, String type) {
        // LuckPerms for permission check, fallback to file for preference
        String node = "dap-ur-homies." + type;
        boolean hasPerm = hasPermission(player, node);
        if (!hasPerm) return false;
        // Also check personal preference (file-based)
        return fallback.canInitiate(player, type);
    }

    @Override
    public boolean canReceive(ServerPlayerEntity player, String type) {
        // Default: file-based preference only (players control their own receive settings)
        if (!ModConfig.get().luckPermsControlsReceive) {
            return fallback.canReceive(player, type);
        }
        // Optional: LuckPerms can override receive permissions via "dap-ur-homies.receive.<type>"
        String node = "dap-ur-homies.receive." + type;
        boolean hasPerm = hasPermission(player, node);
        if (!hasPerm) return false;
        // Still respect personal file preference as final gate
        return fallback.canReceive(player, type);
    }

    @Override
    public void setPreference(ServerPlayerEntity player, String type, boolean value) {
        fallback.setPreference(player, type, value);
    }

    @Override
    public PlayerPreferences getPerms(ServerPlayerEntity player) {
        return fallback.getPerms(player);
    }

    @Override
    public PlayerPreferences getPerms(java.util.UUID playerId) {
        return fallback.getPerms(playerId);
    }

    @Override
    public String getLang(java.util.UUID playerId) {
        return fallback.getLang(playerId);
    }

    @Override
    public void setLang(java.util.UUID playerId, String lang) {
        fallback.setLang(playerId, lang);
    }
}

package com.dap.dapurhomies.util;

import com.google.gson.Gson;
import com.google.gson.GsonBuilder;
import net.fabricmc.loader.api.FabricLoader;

import java.io.Reader;
import java.io.Writer;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Set;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

/**
 * Admin control: temporary permission bans + custom cooldown overrides + admin-forced disable.
 * Bans are wall-clock based so they survive server restarts; cooldown overrides are ticks.
 * Admin bypasses everything.
 */
public class AdminManager {
    public static final long MIN_BAN_TICKS = 1200;        // 1 minute
    public static final long MIN_COOLDOWN_TICKS = 0;      // 0 = no cooldown
    public static final Set<String> VALID_TYPES = Set.of(
            "all", "dap", "highFive", "grab", "push", "kick", "hug", "slap", "clap", "danse");

    // temp bans: key = uuid|type ("all" = total ban), value = epoch millis when it expires
    private static final Map<String, Long> bans = new ConcurrentHashMap<>();
    // global cooldown overrides: type -> ticks (applies to everyone)
    private static final Map<String, Long> globalCooldown = new ConcurrentHashMap<>();
    // per-player cooldown overrides: uuid|type -> ticks
    private static final Map<String, Long> playerCooldown = new ConcurrentHashMap<>();
    // players an admin explicitly disabled; the player cannot undo this with /dap on
    private static final Set<String> forceDisabled = ConcurrentHashMap.newKeySet();

    private static final Gson GSON = new GsonBuilder().setPrettyPrinting().create();
    private static final Path FILE = FabricLoader.getInstance().getConfigDir()
            .resolve("dap-ur-homies-admin.json");

    static {
        load();
    }

    private static String key(UUID id, String type) { return id.toString() + "|" + type; }

    public static boolean isValidType(String type) {
        return type != null && VALID_TYPES.contains(type);
    }

    // ---------- PERSISTENCE ----------
    private static final class Data {
        Map<String, Long> bans = new LinkedHashMap<>();
        Map<String, Long> globalCooldown = new LinkedHashMap<>();
        Map<String, Long> playerCooldown = new LinkedHashMap<>();
        Set<String> forceDisabled = new java.util.LinkedHashSet<>();
    }

    private static synchronized void load() {
        if (!Files.exists(FILE)) return;
        try (Reader r = Files.newBufferedReader(FILE, StandardCharsets.UTF_8)) {
            Data d = GSON.fromJson(r, Data.class);
            if (d == null) return;
            if (d.bans != null) bans.putAll(d.bans);
            if (d.globalCooldown != null) globalCooldown.putAll(d.globalCooldown);
            if (d.playerCooldown != null) playerCooldown.putAll(d.playerCooldown);
            if (d.forceDisabled != null) forceDisabled.addAll(d.forceDisabled);
        } catch (Throwable t) {
            // Corrupt admin file must never prevent the server from starting
            com.dap.dapurhomies.DapUrHomiesMod.LOGGER.warn(
                    "[DapUrHomies] Failed to read admin data, starting with empty state: " + t.getMessage());
        }
    }

    private static synchronized void save() {
        try {
            Data d = new Data();
            d.bans.putAll(bans);
            d.globalCooldown.putAll(globalCooldown);
            d.playerCooldown.putAll(playerCooldown);
            d.forceDisabled.addAll(forceDisabled);
            Files.createDirectories(FILE.getParent());
            try (Writer w = Files.newBufferedWriter(FILE, StandardCharsets.UTF_8)) {
                GSON.toJson(d, w);
            }
        } catch (Throwable t) {
            com.dap.dapurhomies.DapUrHomiesMod.LOGGER.warn(
                    "[DapUrHomies] Failed to save admin data: " + t.getMessage());
        }
    }

    // ---------- BANS ----------
    public static void ban(UUID id, String type, long ticks) {
        long t = Math.max(ticks, MIN_BAN_TICKS);
        bans.put(key(id, type), System.currentTimeMillis() + t * 50L);
        save();
    }

    public static void unban(UUID id, String type) {
        bans.remove(key(id, type));
        save();
    }

    /** true if this player is currently banned from 'type' (or "all"). */
    public static boolean isBanned(UUID id, String type) {
        long now = System.currentTimeMillis();
        Long all = bans.get(key(id, "all"));
        if (all != null) {
            if (now < all) return true;
            bans.remove(key(id, "all"));
        }
        Long t = bans.get(key(id, type));
        if (t != null) {
            if (now < t) return true;
            bans.remove(key(id, type));
        }
        return false;
    }

    // ---------- ADMIN-FORCED DISABLE ----------
    public static void setForceDisabled(UUID id, boolean disabled) {
        if (disabled) forceDisabled.add(id.toString());
        else forceDisabled.remove(id.toString());
        save();
    }

    public static boolean isForceDisabled(UUID id) {
        return forceDisabled.contains(id.toString());
    }

    // ---------- COOLDOWNS ----------
    /** Set a global cooldown override for a module (ticks, floor applied). */
    public static void setGlobalCooldown(String type, long ticks) {
        globalCooldown.put(type, Math.max(ticks, MIN_COOLDOWN_TICKS));
        save();
    }

    /** Set a per-player cooldown override for a module. */
    public static void setPlayerCooldown(UUID id, String type, long ticks) {
        playerCooldown.put(key(id, type), Math.max(ticks, MIN_COOLDOWN_TICKS));
        save();
    }

    public static void clearGlobalCooldown(String type) {
        globalCooldown.remove(type);
        save();
    }

    public static void clearPlayerCooldown(UUID id, String type) {
        playerCooldown.remove(key(id, type));
        save();
    }

    /** Resolve effective cooldown in ticks: player override > global override > config default. */
    public static long resolveCooldownTicks(UUID id, String type, long configMs) {
        Long p = playerCooldown.get(key(id, type));
        if (p != null) return p;
        Long g = globalCooldown.get(type);
        if (g != null) return g;
        return TickClock.msToTicks(configMs);
    }

package com.dap.dapurhomies.util;

import net.minecraft.server.MinecraftServer;
import net.minecraft.server.world.ServerWorld;
import net.minecraft.sound.SoundCategory;
import net.minecraft.sound.SoundEvents;
import net.minecraft.util.math.Vec3d;

import java.util.ArrayList;
import java.util.Iterator;
import java.util.List;

/**
 * Looping clap sound effect for high-five and dap interactions.
 * Plays a rhythmic clapping sound pattern over 2 seconds (40 ticks).
 * Uses snare drum sounds with varying pitch to simulate realistic clapping.
 */
public class ClapLooper {
    private static class Loop {
        ServerWorld w;
        Vec3d pos;
        int left;      // total ticks remaining
        int tickCounter; // counts down to next clap
        float basePitch; // base pitch for this clap sequence
        int rhythmIndex; // current position in rhythm pattern (per-loop, not global)
        int interval;    // configurable spacing offset (5 = original rhythm)
    }
    private static final List<Loop> loops = new ArrayList<>();

    // Clap rhythm pattern: ticks between each clap (creates a natural clapping rhythm)
    // Pattern: fast-fast-slow-fast-fast-slow = like real applause
    private static final int[] CLAP_RHYTHM = {2, 3, 2, 3, 2, 4, 2, 3, 2, 3, 2, 4};
    private static final float[] CLAP_PITCH = {1.0f, 1.1f, 0.9f, 1.2f, 1.0f, 0.85f, 1.15f, 1.0f, 1.05f, 0.95f, 1.1f, 0.9f};
    private static final float[] CLAP_VOLUME = {0.8f, 0.9f, 0.7f, 1.0f, 0.8f, 0.6f, 0.95f, 0.85f, 0.75f, 0.9f, 0.8f, 0.7f};

    public static void start(ServerWorld w, Vec3d pos, int totalTicks, int interval) {
        Loop l = new Loop();
        l.w = w;
        l.pos = pos;
        l.left = Math.max(totalTicks, 1);
        l.tickCounter = 0; // play immediately on first tick
        l.basePitch = 1.0f;
        l.interval = Math.max(1, interval);
        loops.add(l);
    }

    public static void tick(MinecraftServer server) {
        Iterator<Loop> it = loops.iterator();
        while (it.hasNext()) {
            Loop l = it.next();
            l.left--;
            if (l.left <= 0) { it.remove(); continue; }

            l.tickCounter--;
            if (l.tickCounter <= 0) {
                // Set next clap timing based on rhythm pattern
                int idx = l.rhythmIndex % CLAP_RHYTHM.length;
                // interval=5 keeps the original rhythm; larger values space claps further apart
                l.tickCounter = Math.max(1, CLAP_RHYTHM[idx] + (l.interval - 5));

                // Play clap sound with varying pitch and volume for realism
                float pitch = CLAP_PITCH[idx % CLAP_PITCH.length];
                float volume = CLAP_VOLUME[idx % CLAP_VOLUME.length];

                // Use snare drum sound for more realistic clapping
                l.w.playSound(null, l.pos.x, l.pos.y, l.pos.z,
                        SoundEvents.BLOCK_NOTE_BLOCK_SNARE, SoundCategory.PLAYERS, volume, pitch);

                l.rhythmIndex++;
            }
        }
    }
}

/**
 * Unified cooldown manager. Replaces scattered cooldown maps in each handler.
 * Thread-safe using ConcurrentHashMap.
 */
public class CooldownManager {
    private static final Map<String, Map<UUID, Long>> cooldowns = new ConcurrentHashMap<>();

    /**
     * Check if a player is on cooldown for a specific action type.
     */
    public static boolean isOnCooldown(UUID playerId, String type) {
        Map<UUID, Long> map = cooldowns.get(type);
        if (map == null) return false;
        Long end = map.get(playerId);
        if (end == null) return false;
        long now = TickClock.now();
        if (now >= end) {
            map.remove(playerId);
            return false;
        }
        return true;
    }

    /**
     * Set cooldown for a player. Duration is in ticks.
     */
    public static void setCooldown(UUID playerId, String type, long durationTicks) {
        cooldowns.computeIfAbsent(type, k -> new ConcurrentHashMap<>())
                .put(playerId, TickClock.now() + durationTicks);
    }

    /**
     * Cleanup all cooldowns for a player (call on disconnect).
     */
    public static void cleanup(UUID playerId) {
        for (Map<UUID, Long> map : cooldowns.values()) {
            map.remove(playerId);
        }
    }

    /**
     * Purge expired cooldowns to prevent memory growth.
     */
    public static void purgeExpired() {
        long now = TickClock.now();
        for (Map<UUID, Long> map : cooldowns.values()) {
            // Remove entries where the end time has been reached (now >= end).
            map.entrySet().removeIf(e -> now >= e.getValue());
        }
    }
}

package com.dap.dapurhomies.util;

import net.minecraft.server.MinecraftServer;

/**
 * Server-tick based clock. Replaces System.currentTimeMillis() for all cooldowns,
 * charges, and timers to avoid issues with system time jumps (NTP, /time add, sleep).
 * 1 tick = 50ms at 20 TPS.
 */
public class TickClock {
    private static long currentTick = 0;

    public static void update(MinecraftServer server) {
        currentTick = server.getTicks();
    }

    public static long now() {
        return currentTick;
    }

    /** Convert milliseconds to ticks (rounded up). */
    public static long msToTicks(long ms) {
        return (ms + 49) / 50;
    }
}
